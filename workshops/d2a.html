"""Run the same parameterized vector query with a direct connection and a pool."""
# Velg setup, query eller compare fra kommandolinjen.
import argparse
# Les tilkoblingen fra miljøet fremfor å lagre passord i filen.
import os
# Mål tiden brukt på direkte forbindelser og forbindelsespool.
import time
# Finn SQL-filen ved siden av dette skriptet.
from pathlib import Path

# Bruk PostgreSQL-driveren til å kjøre SQL.
import psycopg
# Registrer hvordan pgvector-verdier konverteres i forbindelsen.
from pgvector.psycopg import register_vector
# Gjenbruk et lite antall databaseforbindelser.
from psycopg_pool import ConnectionPool


# Kjør dette oppsettet på hver nye forbindelse i poolen.
def configure(connection):
    # Lær forbindelsen å håndtere databasens vector-type.
    register_vector(connection)
    # Avslutt transaksjonen fra typeoppsettet før forbindelsen brukes av poolen.
    connection.commit()


# Bruk samme spørring i begge målingene.
def query(connection):
    # Kjør parameterisert SQL og hent de to nærmeste radene.
    return connection.execute(
        # Hent dokumentets ID, metadata og tekst.
        "SELECT id, topic, content FROM workshop_notes "
        # Filtrer på emne og sorter etter cosinusavstand med pgvector.
        "WHERE topic = %s ORDER BY embedding <=> %s::vector LIMIT 2",
        # Send verdiene separat fra SQL-teksten.
        ("containers", "[0,1,0]"),
    ).fetchall()


# Samle oppsett, spørring og sammenligning i ett kjørbart eksempel.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Tillat bare de tre laboperasjonene.
    parser.add_argument("mode", choices=["setup", "query", "compare"])
    # Les operasjonen deltakeren valgte.
    args = parser.parse_args()
    # Bruk tilkoblingsstrengen satt i terminalen.
    connection_string = os.environ["PGCONNINFO"]
    # Disable prepared statements so the example also works through transaction pooling.
    # Deaktiver automatisk server-side prepare for å støtte transaction pooling.
    options = {"prepare_threshold": None}
    # Opprett skjema og eksempeldata.
    if args.mode == "setup":
        # Åpne og lukk en direkte forbindelse for denne operasjonen.
        with psycopg.connect(connection_string, **options) as connection:
            # Les setup.sql fra samme mappe og kjør innholdet på serveren.
            connection.execute(Path(__file__).with_name("setup.sql").read_text(encoding="utf-8"))
        # Bekreft at oppsettet er fullført.
        print("Schema and four example rows ready")
    # Kjør ett vanlig oppslag.
    elif args.mode == "query":
        # Åpne og lukk en direkte forbindelse for denne operasjonen.
        with psycopg.connect(connection_string, **options) as connection:
            # Behandle radene fra den parameteriserte spørringen.
            for row in query(connection):
                # Vis resultatet i terminalen.
                print(row)
    # Sammenlign tidsbruk med og uten forbindelsespool.
    else:
        # Bruk samme antall operasjoner i begge målingene.
        count = 10
        # Start tidsmålingen for direkte forbindelser.
        started = time.perf_counter()
        # Åpne en ny forbindelse for hver repetisjon.
        for _ in range(count):
            # Inkluder forbindelsesopprettelsen i målingen.
            with psycopg.connect(connection_string, **options) as connection:
                # Kjør samme arbeid som i pool-varianten.
                query(connection)
        # Vis samlet tid for nye forbindelser.
        print(f"{count} new connections: {time.perf_counter() - started:.3f}s")
        # Opprett en pool med mellom én og tre forbindelser.
        with ConnectionPool(connection_string, min_size=1, max_size=3,
                            # Bruk samme tilkoblingsvalg og registrer pgvector for hver forbindelse.
                            kwargs=options, configure=configure) as pool:
            # Vent til poolen er klar før målingen starter.
            pool.wait()
            # Start målingen etter klargjøringen av poolen.
            started = time.perf_counter()
            # Gjenta samme antall spørringer.
            for _ in range(count):
                # Lån en forbindelse og lever den tilbake etter bruk.
                with pool.connection() as connection:
                    # Kjør den samme spørringen på en gjenbrukt forbindelse.
                    query(connection)
            # Vis tiden for spørringene gjennom poolen.
            print(f"{count} pooled queries: {time.perf_counter() - started:.3f}s")


# Start bare når filen kjøres direkte.
if __name__ == "__main__":
    # Kjør den valgte laboperasjonen.
    main()

Which pgvector operator and index operator class implement Euclidean distance?

Use <-> for Euclidean, or L2, distance and vector_l2_ops for the matching index. L2 is straight-line distance: between [1,2] and [4,6], it is 5. ORDER BY embedding <-> query_vector LIMIT 5 selects the five smallest distances.

m-26-1-1 · Åpne dette flashcardet

Which pgvector operator and index operator class implement cosine distance?

Use <=> with vector_cosine_ops. Cosine distance is 1 minus cosine similarity, so lower values mean more similar directions. For nonzero vectors, zero means the same direction. Order the operator result ascending and apply LIMIT for nearest-neighbor retrieval.

Eksempel · PostgreSQL · pgvector

SELECT '[1,0]'::vector <=> '[2,0]'::vector;

With pgvector enabled, the result is 0: these nonzero vectors point in the same direction. Cosine compares direction, so different lengths alone do not make them distant.

m-26-2-1 · Åpne dette flashcardet

Which pgvector operator and index operator class implement negative inner product?

Use <#> with vector_ip_ops for the matching index. An inner product (dot product) multiplies matching vector components and adds them: [1,2] · [3,4] = 1×3 + 2×4 = 11. Negative inner product means negating that sum, so <#> returns -11. If the dot product were -2, it would return +2. Inner-product search ranks larger dot products first. Negating them makes this work with ascending index ordering: -11 sorts before -5, so a dot product of 11 ranks above 5. Example: ORDER BY embedding <#> '[1,2]' LIMIT 5. The dot product depends on both vector direction and length. For vectors normalized to length 1, it equals cosine similarity.

m-26-3-1 · Åpne dette flashcardet

Why must a pgvector query operator match its index operator class?

The operator class determines which distance ordering an index supports. An index built with vector_cosine_ops supports cosine distance through <=>; it does not supply the L2 ordering requested by <->. Matching them makes the index eligible. Use EXPLAIN to see whether the planner actually chooses it.

How do index choices differ for embeddings and ordinary filter columns in PostgreSQL?

Use a vector index, such as HNSW or IVFFlat, for embedding similarity search. Use suitable ordinary indexes, such as B-tree, for metadata predicates like a department or date range. Choose indexes from the actual query pattern and check the execution plan rather than assuming that every index will be used together.

Eksempel · PostgreSQL

CREATE INDEX docs_department ON documents (department);
CREATE INDEX docs_vectors ON documents USING hnsw (embedding vector_cosine_ops);

For existing department and vector embedding columns, the first index supports ordinary metadata predicates and the second supports cosine-vector ordering. The planner chooses the access path.

How should user input be passed to a PostgreSQL query?

Pass values as driver parameters, separately from the SQL text. With Psycopg: cur.execute("SELECT content FROM documents WHERE id = %s", (document_id,)) The driver binds document_id as data. Keep %s unquoted, and do not use Python string formatting to insert the value into the SQL.

Which PostgreSQL type stores an embedding with a fixed dimension count?

Use pgvector’s vector(n), where n is the component count. For example, vector(1536) stores a 1,536-dimensional embedding. Query vectors and the chosen index must be compatible with that dimension count and representation.

In what order should you load data and build indexes for PostgreSQL vector search?

Create the table, load representative data, build the indexes and run ANALYZE before benchmarking queries. Loading first avoids maintaining each secondary index during every insert. Representative data is particularly important for IVFFlat because it trains its lists from the existing vectors.

What SQL retrieves the five closest Finance documents by cosine distance?

Filter the metadata, order by cosine distance and limit the results: SELECT id FROM documents WHERE department = 'Finance' ORDER BY embedding <=> $1::vector LIMIT 5; Bind $1 to the query embedding using the database client; Psycopg uses its own %s placeholder convention. With an approximate index, filtering may leave fewer matches, so inspect the plan and tune search effort or supported iterative scans when needed.

How can a B-tree index help a metadata-filtered PostgreSQL vector query?

A B-tree index can make an ordinary predicate, such as department = 'Finance', cheaper to evaluate. It does not calculate vector similarity. The planner decides whether to filter candidates first or use a vector-index path. Inspect EXPLAIN; approximate vector scans may apply filters after selecting candidates.

This can help select Finance rows before other work where the chosen plan uses it. It is not itself a nearest-neighbor index; verify filtered-vector plans with EXPLAIN.

Which index belongs on a PostgreSQL embedding column?

Choose HNSW or IVFFlat with the operator class matching the query metric. A cosine query using <=> needs vector_cosine_ops. Benchmark index memory, build time, query latency and recall on representative data before choosing an algorithm and settings.

Eksempel · PostgreSQL · pgvector

CREATE INDEX docs_cosine_idx ON documents USING hnsw (embedding vector_cosine_ops);

This example matches cosine queries using <=> on an existing embedding column. Choose and benchmark the algorithm and operator class for the real workload.

m-29-5-1 · Åpne dette flashcardet

What does EXPLAIN ANALYZE measure that EXPLAIN alone does not?

EXPLAIN shows the planner’s estimates. EXPLAIN ANALYZE executes the statement and reports observed timing and row counts. A modifying statement really changes data when executed. Use an appropriate test environment or a deliberately rolled-back transaction when analyzing such a statement.

Eksempel · PostgreSQL

EXPLAIN (ANALYZE, BUFFERS) SELECT id FROM documents WHERE department = 'Finance';

This executes the SELECT on an existing table and reports observed timings, rows and buffer activity. A modifying statement inside EXPLAIN ANALYZE would also execute its modifications.

How does a client-side PostgreSQL connection pool reduce overhead?

The pool lends a request an already-established connection, avoiding repeated network setup, TLS negotiation and authentication. Return the connection after use so another request can borrow it. A bounded pool also controls concurrent connection usage.

Eksempel · Scenario

The first request opens a database connection and returns it to the pool. A later request borrows that established connection, avoiding a second network/TLS/authentication setup for the same server session.

m-28-2-1 · Åpne dette flashcardet

How should a PostgreSQL connection pool be bounded across app instances?

Plan the total across the maximum application instance count, leaving room for other clients and administration. A 500-connection app budget across ten instances permits at most 50 connections per instance. Setting each instance’s pool to 500 would instead permit 5,000 combined connections.

m-28-3-1 · Åpne dette flashcardet

How can Psycopg return a borrowed PostgreSQL connection to its pool automatically?

Borrow the connection through the pool’s context manager: with pool.connection() as conn: with conn.cursor() as cur: cur.execute("SELECT 1") Here pool is an existing Psycopg ConnectionPool. Leaving the connection block returns it to the pool, including when an exception leaves the block.

"""A real PostgreSQL source with a Redis cache; no distributed transaction."""
# Les operasjon og labdokument fra kommandolinjen.
import argparse
# Serialiser den offentlige testteksten i cacheverdien.
import json
# Les tilkoblinger fra terminalen.
import os


# Hent ett dokument og returner også hvor svaret kom fra.
def read_document(cache, connection, document_id):
    # Bruk et eget prefiks for denne nye laben.
    key = f"ai200:v2:document:{document_id}"
    # Forsøk cache før datakilden.
    cached = cache.get(key)
    # En tom streng er også en eksisterende verdi, så test eksplisitt mot None.
    if cached is not None:
        # Returner den serialiserte teksten uten databaseoppslag.
        return json.loads(cached), "hit"
    # Send ID som parameter fremfor å bygge SQL fra input.
    row = connection.execute("SELECT content FROM workshop_notes WHERE id = %s", (document_id,)).fetchone()
    # Ikke cache et fraværende dokument i denne enkle øvelsen.
    if row is None:
        # Gjør fravær forskjellig fra et tomt innhold.
        raise LookupError(f"No lab document {document_id}")
    # Cache kildeverdien i tretti sekunder.
    cache.set(key, json.dumps(row[0]), ex=30)
    # Returner kildedata og vis at dette var en miss.
    return row[0], "miss"


# Oppdater datakilden før den gamle cacheverdien fjernes.
def update_document(cache, connection, document_id, text):
    # Oppdater bare dokumentet deltakeren valgte.
    cursor = connection.execute("UPDATE workshop_notes SET content = %s WHERE id = %s", (text, document_id))
    # En ukjent ID skal ikke se ut som en vellykket endring.
    if cursor.rowcount != 1:
        # Avbryt hvis dokumentet ikke finnes.
        raise LookupError(f"No lab document {document_id}")
    # Gjør kildeendringen varig før invalidering; dette er fortsatt ikke atomisk med Redis.
    connection.commit()
    # Fjern kun denne labnøkkelen.
    cache.delete(f"ai200:v2:document:{document_id}")


# Kjør en eksplisitt lese- eller skriveoperasjon.
def main():
    # Beskriv kommandolinjen.
    parser = argparse.ArgumentParser()
    # Tillat bare labens to operasjoner.
    parser.add_argument("mode", choices=["read", "update"])
    # Standarddokumentet finnes etter setup.sql.
    parser.add_argument("--id", type=int, default=1)
    # Krev en ny tekst bare ved oppdatering.
    parser.add_argument("--text")
    # Les valgene før eksterne biblioteker og tjenester brukes.
    args = parser.parse_args()
    # Avvis en ufullstendig skrivekommando.
    if args.mode == "update" and args.text is None:
        # Gi en konkret kommandolinjefeil.
        parser.error("update requires --text")
    # Importer databasedriveren for den faktiske kjøringen.
    import psycopg
    # Importer Redis-klienten for den faktiske kjøringen.
    import redis
    # Åpne begge forbindelser og lukk dem etter operasjonen.
    with psycopg.connect(os.environ["PGCONNINFO"]) as connection, redis.Redis(
        # Bruk Managed Redis-endpointet fra din ressurs.
        host=os.environ["REDIS_HOST"], port=int(os.getenv("REDIS_PORT", "10000")),
        # Les nøkkelen lokalt, bruk TLS og avgrens nettverksventingen.
        password=os.environ["REDIS_KEY"], ssl=True, socket_timeout=10
    ) as cache:
        # Les gjennom cache-aside-funksjonen.
        if args.mode == "read":
            # Vis bare den ufarlige labteksten og hvor den ble hentet.

import argparse
# Les Redis-endpoint og labnøkkel fra terminalens miljø.
import os
# Vent på en kort TTL for å demonstrere utløp.
import time
# Konverter spørsmåls- og dokumentvektorer til float32-byteverdier.
from array import array

# Bruk Redis-klienten til cache- og søkekommandoer.
import redis
# Beskriv metadata, tekst og vektorer i søkeindeksen.
from redis.commands.search.field import TagField, TextField, VectorField
# Avgrens hvilke Redis-hasher indeksen leser.
from redis.commands.search.index_definition import IndexDefinition, IndexType
# Bygg et søk med filter, KNN og resultatvalg.
from redis.commands.search.query import Query
# Gjenkjenn feilsvaret når indeksen allerede finnes.
from redis.exceptions import ResponseError


# Kjør én Redis-operasjon.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Begrens operasjonene til labens fire valg.
    parser.add_argument("mode", choices=["cache", "index", "search", "cleanup"])
    # Les valgt operasjon.
    args = parser.parse_args()
    # Bruk endpoint og port for labinstansen.
    with redis.Redis(host=os.environ["REDIS_HOST"], port=int(os.getenv("REDIS_PORT", "10000")),
                     # Bruk TLS, miljøbasert labnøkkel og en avgrenset nettverkstimeout.
                     password=os.environ["REDIS_KEY"], ssl=True, socket_timeout=10) as client:
        # Demonstrer hit, invalidering og TTL-utløp.
        if args.mode == "cache":
            # Bruk en avgrenset nøkkel for labens cacheoppføring.
            key = "ai200:cache:containers:v1"
            # Kontroller om oppføringen finnes fra før.
            print("Initial:", client.get(key))
            # Cache resultatet med ti sekunders TTL.
            client.set(key, "note-1,note-2", ex=10)
            # Les den nye oppføringen og gjenværende levetid.
            print("Hit:", client.get(key), "TTL:", client.ttl(key))
            # Invalider oppføringen eksplisitt.
            client.delete(key)
            # Bekreft at oppslaget ikke lenger gir en cacheverdi.
            print("Invalidated:", client.get(key))
            # Legg inn en oppdatert verdi med kort TTL.
            client.set(key, "updated-note-1", ex=2)
            # Vent lenger enn levetiden.
            time.sleep(3)
            # Vis at Redis har latt oppføringen utløpe.
            print("Expired:", client.get(key), "TTL:", client.ttl(key))
        # Opprett en vektorindeks og tre dokumenter.
        elif args.mode == "index":

            fields = [TagField("topic"), TextField("text"), VectorField("embedding", "HNSW", {

                "TYPE": "FLOAT32", "DIM": 3, "DISTANCE_METRIC": "COSINE",

            try:
                # Opprett søkeindeksen med det valgte skjemaet.
                client.ft("ai200-index").create_index(fields, definition=IndexDefinition(
                    # Les bare Redis-hasher med labens vektorprefiks.
                    prefix=["ai200:vector:"], index_type=IndexType.HASH))
            # Undersøk hvorfor indeksopprettelsen ble avvist.
            except ResponseError as error:
                # Ignorer bare den forventede feilen om en eksisterende indeks.
                if "index already exists" not in str(error).lower():
                    # La andre feil gå videre til terminalen.
                    raise
            # Første dokument ligger langs første vektordimensjon.
            for number, topic, vector in [(1, "containers", [1, 0, 0]),
                                           # Legg til et lignende dokument og et dokument i et annet emne.
                                           (2, "containers", [0.9, 0.1, 0]), (3, "data", [0, 1, 0])]:

How does Redis expiration differ from eviction?

Expiration makes a key unavailable after its configured expiry deadline, even when Redis has plenty of free memory. For example, SET greeting hello EX 60 gives the value a 60-second lifetime. Eviction removes eligible keys to free memory when the configured memory limit is reached. It can remove a key before its expiry, or remove a key with no expiry if the policy allows it. The eviction policy determines which keys can be discarded.

How does the cache-aside pattern work?

Read the cache first. On a miss, read the source database, save a cached copy with an appropriate lifetime and return the result. Later hits can reuse it. After a successful source update, invalidate or refresh the related cached value so future requests can obtain fresh data.

Eksempel · Scenario

The first request for repair 42 misses Redis, reads its status from Cosmos DB and caches it. The next request can use that cached value. After a successful status update, invalidate the cached entry so later reads can reload it.

How can a Redis cache entry be given a fixed lifetime?

Give a string a lifetime when writing it, for example SET greeting hello EX 600 for ten minutes. For an existing hash, EXPIRE doc:1 600 sets the whole hash’s lifetime. The clock starts when expiry is set. Setting it again resets that lifetime, so do this only when the application intends to extend or change it.

How should cached data be invalidated after its source changes?

In a cache-aside application, update the source database successfully, then delete or replace the corresponding cached value. For example, DEL customer:42 makes the next cache miss reload that customer from the database. If other applications can change the source, arrange change notifications or use an expiry policy to limit how long old data can remain. Saving Redis data to disk preserves cached values for recovery; it does not refresh them from the source. Use known cache keys where possible. SCAN can discover keys incrementally without the blocking whole-keyspace scan performed by KEYS.

How can a Redis string’s expiry be refreshed whenever it is read?

GETEX reads a string value and can reset its expiry in the same operation: GETEX session:42 EX 600 A successful read gives that key another ten minutes. If it is not read again, it expires. This implements sliding expiration for that string key.

Which vector distance metrics does RediSearch support?

The supported choices are COSINE, L2 and IP: cosine, Euclidean and inner-product distance. Set DISTANCE_METRIC when creating the vector field’s index. Redis search returns distance scores, so smaller scores rank closer matches.

Eksempel · Redis vector-field option

DISTANCE_METRIC COSINE

Use this option inside an FT.CREATE vector field definition. With distance scores such as 0.08 and 0.35, the smaller score ranks closer; a distance score is not a larger-is-better similarity score.

How do you query Redis for the five closest embedding vectors?

With index documents, a connected redis-py client r and a correctly encoded query_blob: from redis.commands.search.query import Query q = Query("*=>[KNN 5 @embedding $query_vec AS score]").sort_by("score").return_fields("score").dialect(2) results = r.ft("documents").search(q, query_params={"query_vec": query_blob}) KNN 5 requests five neighbors. @embedding selects the vector field, and query_vec supplies the query bytes. Ordering score ascending puts the closest distance scores first; query bytes must match the field’s type and dimensions.

How do Redis volatile-* and allkeys-* eviction policies differ?

volatile-* policies can evict only keys that have an expiry. allkeys-* policies can evict keys with or without an expiry. For example, volatile-lru chooses among expiring keys, while allkeys-lru considers all keys. If no keys have an expiry, a volatile policy has no eviction candidates and writes needing more memory can fail. Redis calls the setting maxmemory-policy; Azure exposes an Eviction Policy setting. Choose a policy based on which cached data may be discarded.

Eksempel · Scenario

Key A has an expiry; key B has none. A volatile policy can choose only A for eviction. An allkeys policy can consider either. These are eligibility rules; the rest of the policy decides how candidates are ranked.

What does Redis noeviction do when memory is exhausted?

NoEviction keeps existing keys instead of discarding them to make room. Writes needing additional memory can fail when the memory limit is reached. Reads and operations that release memory can continue, and normal time-based expiration still applies.

Eksempel · Scenario

A full NoEviction cache still holds repair:42. Reading that key can succeed, while adding a new key that needs more memory fails. NoEviction does not make all Redis commands stop working.

How do Redis LRU and LFU rank keys for eviction?

LRU, Least Recently Used, favors keys that have not been accessed recently. LFU, Least Frequently Used, favors keys used less often. Redis approximates these choices, and LFU reduces the influence of older frequency history over time.

Eksempel · Scenario

A was read just now but is rarely used; B was used frequently but less recently. Recency favors keeping A, while frequency can favor keeping B. Redis’s approximate sampling and frequency aging affect the real choice.

How do Redis FLAT and HNSW vector indexes differ?

FLAT compares the candidate vectors exhaustively to find exact neighbors. HNSW navigates a graph approximately, which can improve large-search performance while using index memory and potentially missing some exact neighbors. Compare recall and latency for the actual dataset.

Eksempel · Redis search commands

FT.CREATE exact ON HASH PREFIX 1 doc: SCHEMA embedding VECTOR FLAT 6 TYPE FLOAT32 DIM 3 DISTANCE_METRIC COSINE
FT.CREATE approximate ON HASH PREFIX 1 doc: SCHEMA embedding VECTOR HNSW 6 TYPE FLOAT32 DIM 3 DISTANCE_METRIC COSINE

These alternative indexes use the same illustrative three-dimensional hash-vector field, enabling an exact-versus-approximate comparison. Vector search must already be supported/configured, and stored bytes must match FLOAT32 and DIM 3.

When you create a Redis search index with FT.CREATE, does it include data already stored in Redis?

Yes, by default. When you create the index, Redis scans data already stored in the database and indexes documents that meet the index's rules. For example, PREFIX 1 doc: limits it to keys whose names start with doc:, such as doc:123. SKIPINITIALSCAN tells Redis to skip that scan of existing data. The scan can still be running after FT.CREATE returns, so use FT.INFO to check its progress.

What combination suggests a Redis cache may be undersized?

Look for sustained memory pressure, frequent evictions and a worsening cache hit ratio together. Inspect key sizes, expiry rules and workload changes before increasing capacity. A poor hit ratio alone can reflect ineffective caching rather than insufficient memory.

How can an application reduce a cache stampede when a popular key expires?

Coordinate refresh so one worker rebuilds a missing popular value while others wait or use an acceptable stale copy. A short-lived lock can support that coordination. Adding a little variation to expiry times also helps prevent many keys from expiring together.

What does Redis PERSIST do?

PERSIST key removes the key’s expiry time. The value can remain until explicitly deleted or otherwise removed, such as by an applicable eviction policy. This command changes expiry; disk durability is configured separately.

Eksempel · Redis commands

PERSIST repair:42
TTL repair:42

Assuming the key exists and remains present, the first command removes its expiry and the second returns -1. This changes lifetime, not whether Redis data is persisted to disk.

What do Redis TTL return values -1 and -2 mean?

TTL reports the remaining lifetime in seconds. A result of -1 means the key exists without an expiry; -2 means it does not exist. For example, 120 means about two minutes remain. These values are useful when diagnosing unexpectedly persistent or missing cache entries.
