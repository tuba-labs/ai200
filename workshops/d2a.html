"""Run the same parameterized vector query with a direct connection and a pool."""
# Velg setup, query eller compare fra kommandolinjen.
import argparse
# Les tilkoblingen fra miljøet fremfor å lagre passord i filen.
import os
# Mål tiden brukt på direkte forbindelser og forbindelsespool.
import time
# Finn SQL-filen ved siden av dette skriptet.
from pathlib import Path

# Bruk PostgreSQL-driveren til å kjøre SQL.
import psycopg
# Registrer hvordan pgvector-verdier konverteres i forbindelsen.
from pgvector.psycopg import register_vector
# Gjenbruk et lite antall databaseforbindelser.
from psycopg_pool import ConnectionPool


# Kjør dette oppsettet på hver nye forbindelse i poolen.
def configure(connection):
    # Lær forbindelsen å håndtere databasens vector-type.
    register_vector(connection)
    # Avslutt transaksjonen fra typeoppsettet før forbindelsen brukes av poolen.
    connection.commit()


# Bruk samme spørring i begge målingene.
def query(connection):
    # Kjør parameterisert SQL og hent de to nærmeste radene.
    return connection.execute(
        # Hent dokumentets ID, metadata og tekst.
        "SELECT id, topic, content FROM workshop_notes "
        # Filtrer på emne og sorter etter cosinusavstand med pgvector.
        "WHERE topic = %s ORDER BY embedding <=> %s::vector LIMIT 2",
        # Send verdiene separat fra SQL-teksten.
        ("containers", "[0,1,0]"),
    ).fetchall()


# Samle oppsett, spørring og sammenligning i ett kjørbart eksempel.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Tillat bare de tre laboperasjonene.
    parser.add_argument("mode", choices=["setup", "query", "compare"])
    # Les operasjonen deltakeren valgte.
    args = parser.parse_args()
    # Bruk tilkoblingsstrengen satt i terminalen.
    connection_string = os.environ["PGCONNINFO"]
    # Disable prepared statements so the example also works through transaction pooling.
    # Deaktiver automatisk server-side prepare for å støtte transaction pooling.
    options = {"prepare_threshold": None}
    # Opprett skjema og eksempeldata.
    if args.mode == "setup":
        # Åpne og lukk en direkte forbindelse for denne operasjonen.
        with psycopg.connect(connection_string, **options) as connection:
            # Les setup.sql fra samme mappe og kjør innholdet på serveren.
            connection.execute(Path(__file__).with_name("setup.sql").read_text(encoding="utf-8"))
        # Bekreft at oppsettet er fullført.
        print("Schema and four example rows ready")
    # Kjør ett vanlig oppslag.
    elif args.mode == "query":
        # Åpne og lukk en direkte forbindelse for denne operasjonen.
        with psycopg.connect(connection_string, **options) as connection:
            # Behandle radene fra den parameteriserte spørringen.
            for row in query(connection):
                # Vis resultatet i terminalen.
                print(row)
    # Sammenlign tidsbruk med og uten forbindelsespool.
    else:
        # Bruk samme antall operasjoner i begge målingene.
        count = 10
        # Start tidsmålingen for direkte forbindelser.
        started = time.perf_counter()
        # Åpne en ny forbindelse for hver repetisjon.
        for _ in range(count):
            # Inkluder forbindelsesopprettelsen i målingen.
            with psycopg.connect(connection_string, **options) as connection:
                # Kjør samme arbeid som i pool-varianten.
                query(connection)
        # Vis samlet tid for nye forbindelser.
        print(f"{count} new connections: {time.perf_counter() - started:.3f}s")
        # Opprett en pool med mellom én og tre forbindelser.
        with ConnectionPool(connection_string, min_size=1, max_size=3,
                            # Bruk samme tilkoblingsvalg og registrer pgvector for hver forbindelse.
                            kwargs=options, configure=configure) as pool:
            # Vent til poolen er klar før målingen starter.
            pool.wait()
            # Start målingen etter klargjøringen av poolen.
            started = time.perf_counter()
            # Gjenta samme antall spørringer.
            for _ in range(count):
                # Lån en forbindelse og lever den tilbake etter bruk.
                with pool.connection() as connection:
                    # Kjør den samme spørringen på en gjenbrukt forbindelse.
                    query(connection)
            # Vis tiden for spørringene gjennom poolen.
            print(f"{count} pooled queries: {time.perf_counter() - started:.3f}s")


# Start bare når filen kjøres direkte.
if __name__ == "__main__":
    # Kjør den valgte laboperasjonen.
    main()

-- Lag en separat HNSW-indeks for euklidsk avstand.
CREATE INDEX IF NOT EXISTS notes_l2_idx ON workshop_notes USING hnsw (embedding vector_l2_ops);
-- Lag en separat HNSW-indeks for negativt indreprodukt.
CREATE INDEX IF NOT EXISTS notes_ip_idx ON workshop_notes USING hnsw (embedding vector_ip_ops);
-- Oppdater statistikken før planene sammenlignes.
ANALYZE workshop_notes;
-- Sammenlign cosinusavstand bare innenfor valgt emne.
SELECT id, topic, embedding <=> '[0,1,0]'::vector AS distance FROM workshop_notes WHERE topic = 'containers' ORDER BY embedding <=> '[0,1,0]'::vector LIMIT 2;
-- Fjern emnefilteret og se hvorfor et dokument fra data nå kan være nærmest.

-- Bruk operatoren som matcher vector_l2_ops.

Which pgvector operator and index operator class implement Euclidean distance?

Use the <-> operator for Euclidean, also called L2, distance. A pgvector index for this operator uses vector_l2_ops. For example, ORDER BY embedding <-> query_vector LIMIT 5 asks for the five smallest L2 distances.

m-26-1-1 · Åpne dette flashcardet

Which pgvector operator and index operator class implement cosine distance?

Use <=> for cosine distance and vector_cosine_ops for the matching pgvector index. Cosine distance compares vector direction; smaller distance means more similar directions. A nearest-neighbor query orders this distance ascending and applies LIMIT.

m-26-2-1 · Åpne dette flashcardet

Which pgvector operator and index operator class implement negative inner product?

Use <#> for negative inner product and vector_ip_ops for the matching index. PostgreSQL’s index ordering uses ascending distance, so pgvector negates the inner product: the largest original inner products become the smallest returned values.

m-26-3-1 · Åpne dette flashcardet

Why must a pgvector query operator match its index operator class?

The index only supports the distance measure it was built for. For example, a cosine index using vector_cosine_ops cannot provide the L2 ordering requested by <->. Matching them lets PostgreSQL consider that index; EXPLAIN shows whether it actually uses it.

m-26-4-1 · Åpne dette flashcardet

How do index choices differ for embeddings and ordinary filter columns in PostgreSQL?

The embedding column needs a suitable vector index, such as HNSW or IVFFlat, for similarity search. Ordinary filter columns, such as department or created_at, may use B-tree indexes for equality/range conditions. Choose each index from the queries using that column.

m-26-7-1 · Åpne dette flashcardet

How should user input be passed to a PostgreSQL query?

Pass data values separately from SQL. With Psycopg: cur.execute("SELECT content FROM documents WHERE id = %s", (document_id,)) The driver binds document_id as a value; it is not concatenated into the SQL syntax. Do not put quotes around %s or use Python string interpolation to insert user values.

m-28-4-1 · Åpne dette flashcardet

Which PostgreSQL type stores an embedding with a fixed dimension count?

vector(n), provided by pgvector, where n is the number of components. For example, vector(1536) stores a 1,536-dimensional embedding. Stored embeddings and query vectors must use compatible dimensions.

m-29-1-1 · Åpne dette flashcardet

In what order should you load data and build indexes for PostgreSQL vector search?

Create the table, load representative data, build the indexes, run ANALYZE, then test the queries. Loading first avoids updating indexes for each inserted row. It also gives IVFFlat representative vectors for training its clusters.

m-29-2-1 · Åpne dette flashcardet

What SQL retrieves the five closest Finance documents by cosine distance?

Assuming documents has id, department and embedding columns: SELECT id FROM documents WHERE department = 'Finance' ORDER BY embedding <=> $1::vector LIMIT 5; Bind $1 to the query embedding through the database client. The equality condition selects Finance documents; <=> orders by cosine distance, and LIMIT 5 returns at most five results. Some drivers use a different parameter placeholder, such as %s in Psycopg.

m-29-3-1 · Åpne dette flashcardet

How can a B-tree index help a metadata-filtered PostgreSQL vector query?

It can find rows matching the ordinary filter, such as department = 'Finance', before calculating vector distances for those candidates. It does not calculate similarity itself. The planner may choose a different order, so inspect EXPLAIN to see whether filtering or vector retrieval happens first.

m-29-4-1 · Åpne dette flashcardet

Which index belongs on a PostgreSQL embedding column?

For pgvector, HNSW or IVFFlat with the operator class matching the query’s distance measure. For cosine distance, that is vector_cosine_ops. Choose and benchmark the algorithm against the dataset size, required recall and latency target.

m-29-5-1 · Åpne dette flashcardet

What does EXPLAIN ANALYZE measure that EXPLAIN alone does not?

EXPLAIN estimates an execution plan without running the statement. EXPLAIN ANALYZE actually executes it and reports observed times and row counts. For a statement that modifies data, those modifications really occur unless you deliberately contain them in a transaction that is rolled back.

How does a client-side PostgreSQL connection pool reduce overhead?

It lends requests already-open database connections. That avoids repeating network setup, TLS negotiation and authentication for every query. Return the connection to the pool after use so another request can reuse it.

m-28-2-1 · Åpne dette flashcardet

How should a PostgreSQL connection pool be bounded across app instances?

Budget connections across the maximum number of app instances, including scale-out, and reserve capacity for administration and other clients. For example, an app budget of 500 connections across 10 instances allows at most 50 per instance. A per-instance pool size of 500 would instead permit 5,000 combined connections.

m-28-3-1 · Åpne dette flashcardet

How can Psycopg return a borrowed PostgreSQL connection to its pool automatically?

Use the pool’s connection context manager: with pool.connection() as conn: with conn.cursor() as cur: cur.execute("SELECT 1") Here pool is an existing Psycopg ConnectionPool. Leaving the block returns the connection to the pool, allowing another request to reuse it.

Which port connects through built-in PgBouncer on Azure PostgreSQL?

6432, using the server’s usual hostname. Built-in PgBouncer must be enabled. Port 5432 connects directly to PostgreSQL instead of going through that pooler.

How is built-in PgBouncer enabled on Azure PostgreSQL?

Set pgbouncer.enabled to true under the server’s Server parameters and save it. Then connect through port 6432. Enabling this setting does not require a PostgreSQL server restart.

What is the default pool mode of Azure's built-in PgBouncer?

Transaction pooling. A client borrows a server connection for a transaction, and PgBouncer can return that connection to the pool when the transaction ends. The next transaction can use a different server connection.

Which Azure PostgreSQL compute tiers support built-in PgBouncer?

General Purpose and Memory Optimized. The Burstable tier does not support Azure’s built-in PgBouncer feature.

Why is increasing max_connections different from adding a connection pool?

max_connections raises the number of database connections that may exist. A connection pool reuses established connections, reducing repeated setup and authentication work. A higher connection limit alone does not provide that reuse.

How can an application authenticate to Azure PostgreSQL with managed identity?

Enable Microsoft Entra authentication, create the managed identity’s database principal and grant the needed database privileges. The app then obtains an Azure PostgreSQL access token and supplies it as the connection password. New connections need a valid token.

Why might adding PostgreSQL vCores fail to speed up one slow vector query?

The query may not be able to use the extra cores, or may still calculate distances for too many rows. Inspect its plan and index usage first. More vCores help only when additional CPU capacity or parallel work addresses the measured bottleneck.

m-30-1-1 · Åpne dette flashcardet

Which resource can help when a vector-index working set no longer fits in memory?

More memory. It can keep frequently used table and index pages cached, reducing disk reads. Check memory pressure and disk I/O, then compare the same workload before and after scaling.

m-30-2-1 · Åpne dette flashcardet

How do typical Azure PostgreSQL tiers differ in memory per vCore?

Memory Optimized provides more memory per vCore than General Purpose. General Purpose commonly provides 4 GiB per vCore; Memory Optimized ratios depend on VM series, currently about 6.75–9.5 GiB. Burstable ratios also vary by size, so use the chosen SKU’s table for exact sizing.

Which Azure PostgreSQL feature handles growing storage needs?

Storage autogrow. In supported storage configurations, it expands allocated storage when free space runs low. This helps avoid filling the database disk; check the chosen storage type’s size and growth limits.

"""A real PostgreSQL source with a Redis cache; no distributed transaction."""
# Les operasjon og labdokument fra kommandolinjen.
import argparse
# Serialiser den offentlige testteksten i cacheverdien.
import json
# Les tilkoblinger fra terminalen.
import os


# Hent ett dokument og returner også hvor svaret kom fra.
def read_document(cache, connection, document_id):
    # Bruk et eget prefiks for denne nye laben.
    key = f"ai200:v2:document:{document_id}"
    # Forsøk cache før datakilden.
    cached = cache.get(key)
    # En tom streng er også en eksisterende verdi, så test eksplisitt mot None.
    if cached is not None:
        # Returner den serialiserte teksten uten databaseoppslag.
        return json.loads(cached), "hit"
    # Send ID som parameter fremfor å bygge SQL fra input.
    row = connection.execute("SELECT content FROM workshop_notes WHERE id = %s", (document_id,)).fetchone()
    # Ikke cache et fraværende dokument i denne enkle øvelsen.
    if row is None:
        # Gjør fravær forskjellig fra et tomt innhold.
        raise LookupError(f"No lab document {document_id}")
    # Cache kildeverdien i tretti sekunder.
    cache.set(key, json.dumps(row[0]), ex=30)
    # Returner kildedata og vis at dette var en miss.
    return row[0], "miss"


# Oppdater datakilden før den gamle cacheverdien fjernes.
def update_document(cache, connection, document_id, text):
    # Oppdater bare dokumentet deltakeren valgte.
    cursor = connection.execute("UPDATE workshop_notes SET content = %s WHERE id = %s", (text, document_id))
    # En ukjent ID skal ikke se ut som en vellykket endring.
    if cursor.rowcount != 1:
        # Avbryt hvis dokumentet ikke finnes.
        raise LookupError(f"No lab document {document_id}")
    # Gjør kildeendringen varig før invalidering; dette er fortsatt ikke atomisk med Redis.
    connection.commit()
    # Fjern kun denne labnøkkelen.
    cache.delete(f"ai200:v2:document:{document_id}")

# Velg cache-, indeks-, søke- eller oppryddingsoperasjon.
import argparse
# Les Redis-endpoint og labnøkkel fra terminalens miljø.
import os
# Vent på en kort TTL for å demonstrere utløp.
import time
# Konverter spørsmåls- og dokumentvektorer til float32-byteverdier.
from array import array

# Bruk Redis-klienten til cache- og søkekommandoer.
import redis
# Beskriv metadata, tekst og vektorer i søkeindeksen.
from redis.commands.search.field import TagField, TextField, VectorField
# Avgrens hvilke Redis-hasher indeksen leser.
from redis.commands.search.index_definition import IndexDefinition, IndexType
# Bygg et søk med filter, KNN og resultatvalg.
from redis.commands.search.query import Query
# Gjenkjenn feilsvaret når indeksen allerede finnes.
from redis.exceptions import ResponseError


# Kjør én Redis-operasjon.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Begrens operasjonene til labens fire valg.
    parser.add_argument("mode", choices=["cache", "index", "search", "cleanup"])
    # Les valgt operasjon.
    args = parser.parse_args()
    # Bruk endpoint og port for labinstansen.
    with redis.Redis(host=os.environ["REDIS_HOST"], port=int(os.getenv("REDIS_PORT", "10000")),
                     # Bruk TLS, miljøbasert labnøkkel og en avgrenset nettverkstimeout.
                     password=os.environ["REDIS_KEY"], ssl=True, socket_timeout=10) as client:
        # Demonstrer hit, invalidering og TTL-utløp.
        if args.mode == "cache":
            # Bruk en avgrenset nøkkel for labens cacheoppføring.
            key = "ai200:cache:containers:v1"
            # Kontroller om oppføringen finnes fra før.
            print("Initial:", client.get(key))
            # Cache resultatet med ti sekunders TTL.
            client.set(key, "note-1,note-2", ex=10)
            # Les den nye oppføringen og gjenværende levetid.
            print("Hit:", client.get(key), "TTL:", client.ttl(key))
            # Invalider oppføringen eksplisitt.
            client.delete(key)
            # Bekreft at oppslaget ikke lenger gir en cacheverdi.
            print("Invalidated:", client.get(key))
            # Legg inn en oppdatert verdi med kort TTL.
            client.set(key, "updated-note-1", ex=2)
            # Vent lenger enn levetiden.
            time.sleep(3)
            # Vis at Redis har latt oppføringen utløpe.
            print("Expired:", client.get(key), "TTL:", client.ttl(key))
        # Opprett en vektorindeks og tre dokumenter.
        elif args.mode == "index":
            # Indekser topic som filter, text som tekst og embedding med HNSW.
            fields = [TagField("topic"), TextField("text"), VectorField("embedding", "HNSW", {
                # Bruk float32, tre dimensjoner og cosinusavstand.
                "TYPE": "FLOAT32", "DIM": 3, "DISTANCE_METRIC": "COSINE",
            })]
            # Tillat at oppsettet kjøres flere ganger.
            try:
                # Opprett søkeindeksen med det valgte skjemaet.
                client.ft("ai200-index").create_index(fields, definition=IndexDefinition(
                    # Les bare Redis-hasher med labens vektorprefiks.
                    prefix=["ai200:vector:"], index_type=IndexType.HASH))
            # Undersøk hvorfor indeksopprettelsen ble avvist.
            except ResponseError as error:
                # Ignorer bare den forventede feilen om en eksisterende indeks.
                if "index already exists" not in str(error).lower():
                    # La andre feil gå videre til terminalen.
                    raise
            # Første dokument ligger langs første vektordimensjon.
            for number, topic, vector in [(1, "containers", [1, 0, 0]),
                                           # Legg til et lignende dokument og et dokument i et annet emne.
                                           (2, "containers", [0.9, 0.1, 0]), (3, "data", [0, 1, 0])]:
                # Lagre hvert dokument i en Redis-hash som indeksen overvåker.
                client.hset(f"ai200:vector:{number}", mapping={
                    # Lagre metadata og tekst sammen med vektoren.
                    "topic": topic, "text": f"Teaching document {number}",
                    # Send embedding som binære float32-verdier.
                    "embedding": array("f", vector).tobytes(),
                })
            # Minn om at søkeindeksen må ha rukket å behandle dataene.
            print("Index and three documents ready; wait for indexing before search")
        # Kombiner metadatafilter med vektorrangering.
        elif args.mode == "search":
            # Finn to nærmeste treff innen topic=containers, med avstanden kalt score.
            query = Query("(@topic:{containers})=>[KNN 2 @embedding $vec AS score]")
            # Sorter etter avstand og bruk søkedialekten som støtter denne syntaksen.
            query = query.sort_by("score").return_fields("topic", "text", "score").dialect(2)
            # Send spørsmålsvektoren i samme binære format som dokumentvektorene.
            result = client.ft("ai200-index").search(query, query_params={"vec": array("f", [0, 1, 0]).tobytes()})
            # Behandle de returnerte treffene.
            for document in result.docs:
                # Vis ID, emne og avstand for sammenligning.
                print(document.id, document.topic, document.score)
        # Rydd bare labens indeks og tilknyttede dokumenter.
        else:
            # Slett indeksen og dokumentene som hører til den.
            client.ft("ai200-index").dropindex(delete_documents=True)
            # Bekreft oppryddingen.
            print("Removed lab index and its documents")

What triggers Redis key expiration?

A key’s expiry deadline. For example, SET greeting hello EX 60 stores hello under greeting for 60 seconds. Redis removes expired data through its expiration mechanisms without needing memory pressure. The lifetime starts when the expiry is set, unless you explicitly change it later.

How does the cache-aside pattern work?

Read the cache first. On a miss, read the source database, cache the result with an appropriate lifetime and return it. Later hits can use the cached copy. When the source changes, invalidate or update that cached copy so it does not stay stale.

How can a Redis cache entry be given a fixed lifetime?

For a string, SET greeting hello EX 600 stores hello for 600 seconds, or ten minutes. For an existing hash, EXPIRE doc:1 600 gives the whole hash a ten-minute lifetime. TTL means time to live. Set the expiry again only when you intend to reset that lifetime.

After updating the source successfully, delete or replace its related cache entry. For example, DEL customer:42 causes the next request to reload that customer. Prefer known cache keys; if discovery is necessary, SCAN iterates keys incrementally without a blocking KEYS scan.

Why can a Redis cache stay stale even when persistence is enabled?

Persistence saves Redis data for recovery; it does not detect changes in the external source database. The application must update/invalidate the cached value, use an expiry policy, or process source-change events.

How do Redis LRU and LFU rank keys for eviction?

LRU, least recently used, favors removing keys that have not been accessed recently. LFU, least frequently used, favors keys accessed infrequently. Redis approximates these rankings, and LFU’s frequency counter decays over time.

How does Redis volatile-ttl choose an eviction candidate?

It favors expiring keys with the shortest remaining TTL. The policy uses that lifetime to choose a key to remove under memory pressure; it does not refresh or extend the lifetime.

What does Redis PERSIST do?

PERSIST key removes that key’s expiry. The key remains stored until explicitly deleted or removed through another mechanism such as an eligible allkeys eviction policy. It does not make the key durable on disk; Redis persistence is a separate feature.

What do Redis TTL return values -1 and -2 mean?

TTL returns the remaining lifetime in seconds. -1 means the key exists but has no expiry; -2 means the key does not exist. For example, TTL session:42 returning 120 means it is scheduled to expire in about two minutes.
