"""Small isolated Cosmos DB lab; credentials are read from the environment."""
# Velg hvilken Cosmos-operasjon eksemplet skal utføre.
import argparse
# Les endpoint, nøkkel og databasenavn fra terminalens miljø.
import os

# Bruk Cosmos SDK-et og en eksplisitt partisjonsnøkkel.
from azure.cosmos import CosmosClient, PartitionKey

# Bruk små undervisningsdokumenter med håndskrevne vektorer.
DOCUMENTS = [
    # Første dokument ligger i partisjonen containers.
    {"id": "note-1", "topic": "containers", "text": "Package an application", "embedding": [1.0, 0.0, 0.0]},
    # Andre dokument ligger i samme partisjon, med en lignende vektor.
    {"id": "note-2", "topic": "containers", "text": "Inspect container logs", "embedding": [0.9, 0.1, 0.0]},
    # Tredje dokument viser hvordan metadatafilter kan utelate et nært vektortreff.
    {"id": "note-3", "topic": "data", "text": "Retrieve stored data", "embedding": [0.0, 1.0, 0.0]},
]


# La SDK-et sende responsheaderne til denne callbacken.
def report(headers, _body):
    # Hent kostnaden i request units fra responsheaderen.
    charge = headers.get("x-ms-request-charge")
    # Skriv bare RU når headeren finnes.
    if charge:
        # Vis grunnlaget for sammenligning av oppslag.
        print("Request charge:", charge, "RU")


# Kjør operasjonen som ble valgt på kommandolinjen.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Begrens valgene til operasjonene workshopene bruker.
    parser.add_argument("mode", choices=["setup", "query", "point", "vector-setup", "vector", "update"])
    # Les valgt operasjon.
    args = parser.parse_args()
    # Åpne klienten med labkontoens endpoint og nøkkel, og lukk den etter bruk.
    with CosmosClient(os.environ["COSMOS_ENDPOINT"], credential=os.environ["COSMOS_KEY"]) as client:
        # Use a database without shared throughput so vector search is supported.
        # Velg databasen; dette oppretter ikke databasen.
        database = client.get_database_client(os.getenv("COSMOS_DATABASE", "workshop"))
        # Hold vektorpolicyen i en egen container.
        name = "vectors" if args.mode.startswith("vector") else "notes"
        # Hent en klient for den valgte containeren.
        container = database.get_container_client(name)
        # Opprett labcontainer og eksempeldata ved setup.
        if args.mode in {"setup", "vector-setup"}:
            # Beskriv den vanlige indekseringspolicyen.
            policy = {
                # Oppdater indeksene automatisk når dokumentene endres.
                "indexingMode": "consistent", "automatic": True,
                # Indekser dokumentfeltene som standard.
                "includedPaths": [{"path": "/*"}],
                # Unnta vektorarrayet fra vanlig feltindeksering.
                "excludedPaths": [{"path": "/embedding/*"}],
            }
            # Legg bare til vektorinnstillinger for vektorcontaineren.
            extra = {}
            # Konfigurer en egen indeks for VectorDistance-søk.
            if args.mode == "vector-setup":
                # Bruk en flat vektorindeks på embedding-feltet.
                policy["vectorIndexes"] = [{"path": "/embedding", "type": "flat"}]
                # Beskriv hvilke vektorer embedding-feltet inneholder.
                extra["vector_embedding_policy"] = {"vectorEmbeddings": [{
                    # Lagre vektorene i embedding og behandle verdiene som float32.
                    "path": "/embedding", "dataType": "float32",
                    # Krev tre dimensjoner og sammenlign med cosinusavstand.
                    "dimensions": 3, "distanceFunction": "cosine",
                }]}
            # Opprett containeren hvis den mangler; eksisterende policy erstattes ikke her.
            container = database.create_container_if_not_exists(
                # Bruk topic som partisjonsnøkkel.
                id=name, partition_key=PartitionKey(path="/topic"),
                # Bruk policyen over og 400 RU/s dedikert throughput.
                indexing_policy=policy, offer_throughput=400, **extra,
            )
            # Skriv inn undervisningsdokumentene.
            for document in DOCUMENTS:
                # Opprett eller erstatt dokumentet, og vis operasjonens RU.
                container.upsert_item(document, response_hook=report)
            # Vis hvilken container som er klargjort.
            print("Ready:", name)
        # Utfør et direkte oppslag uten SQL-spørring.
        elif args.mode == "point":
            # ID og partisjonsnøkkel identifiserer dokumentet sammen.
            result = container.read_item("note-1", partition_key="containers", response_hook=report)
            # Vis dokumentets ID og tekst.
            print(result["id"], result["text"])
        # Lag en endring som change feed-triggeren kan observere.
        elif args.mode == "update":
            # Les dokumentet før oppdateringen.
            document = container.read_item("note-1", partition_key="containers")
            # Øk et synlig revisjonsnummer.
            document["revision"] = document.get("revision", 0) + 1
            # Lagre den nye versjonen og vis skrivekostnaden.
            container.upsert_item(document, response_hook=report)
            # Bruk revisjonen til å kjenne igjen endringen i triggerloggen.
            print("Updated note-1, revision", document["revision"])
        # Kjør vanlig SQL-søk eller vektorsøk.
        else:
            # Start med et metadatafilter.
            sql = "SELECT c.id, c.topic FROM c WHERE c.topic = @topic"
            # Send filterverdien som en parameter.
            parameters = [{"name": "@topic", "value": "containers"}]
            # Bytt til rangering etter vektoravstand.
            if args.mode == "vector":
                # Returner de to nærmeste treffene og deres avstand.
                sql = ("SELECT TOP 2 c.id, VectorDistance(c.embedding, @vector) AS distance "
                       # Behold emnefilteret og sorter de gjenværende dokumentene etter avstand.
                       "FROM c WHERE c.topic = @topic ORDER BY VectorDistance(c.embedding, @vector)")
                # Bruk samme tredimensjonale spørsmålsvektor som i PostgreSQL-øvelsen.
                parameters.append({"name": "@vector", "value": [0.0, 1.0, 0.0]})
            # Kjør den parameteriserte spørringen.
            results = container.query_items(sql, parameters=parameters,
                                            # Avgrens lesingen til containers-partisjonen og vis RU.
                                            partition_key="containers", response_hook=report)
            # Iterer resultatene fra SDK-et.
            for result in results:
                # Vis hvert treff i terminalen.
                print(result)


# Kjør bare når filen startes direkte.
if __name__ == "__main__":
    # Start den valgte laboperasjonen.
    main()

How many request units does a typical 1-KB Cosmos DB point read cost?

About 1 RU for a direct read by id and partition key under Session, Consistent prefix or Eventual consistency. Strong and Bounded staleness reads cost twice as much, about 2 RU for that item size.

How do you create a Cosmos DB client in Python?

Use the account endpoint and a credential: from azure.cosmos import CosmosClient from azure.identity import DefaultAzureCredential client = CosmosClient(endpoint, credential=DefaultAzureCredential()) Here endpoint is the Cosmos DB account URL. Grant the identity the required Cosmos DB data role and reuse the client across requests.

Which Cosmos DB SDK object performs item CRUD and queries?

The container client, called ContainerProxy in Python. For example, database.get_container_client("items") returns the client used for create_item, read_item, upsert_item, delete_item and query_items. These are item-level operations: create, read, update, delete and query.

What is the Python SDK sequence for querying an existing Cosmos DB container?

Assuming the database and container already exist, and endpoint is the account URL: from azure.cosmos import CosmosClient from azure.identity import DefaultAzureCredential client = CosmosClient(endpoint, credential=DefaultAzureCredential()) database = client.get_database_client("catalog") container = database.get_container_client("items") for item in container.query_items(query="SELECT * FROM c", enable_cross_partition_query=True): print(item) Replace catalog and items with your names. The signed-in identity needs a Cosmos DB data-plane role. Getting these clients selects existing resources; it does not create a database or container.

Should Python Cosmos DB code construct DatabaseProxy and ContainerProxy directly?

No. Obtain them from their parent clients: database = client.get_database_client("catalog") container = database.get_container_client("items") The parent supplies the connection and configuration those proxy objects need.

m-23-5-1 · Åpne dette flashcardet

What two goals should a Cosmos DB partition key balance?

Even distribution and efficient access. A good key spreads writes/storage across many values while letting common reads target a known partition. For example, customerId can work well when traffic is spread across customers; a single popular value can create a hot partition.

prep-cosmos-006 · Åpne dette flashcardet

What happens when a Cosmos DB query cannot be routed to one partition?

The query can fan out to multiple physical partitions and combine their results. That adds coordination and often consumes more request units than a query routed to one partition. Filtering on a known partition-key value helps routing; the exact cost still depends on indexes, data and the query.

What distinguishes a Cosmos DB flat vector index?

It compares all candidate vectors at their configured precision to find exact nearest matches, giving 100% recall. It supports up to 505 dimensions. It is exhaustive rather than an approximate graph search.

m-19-1-1 · Åpne dette flashcardet

Which distance functions can a Cosmos DB vector embedding policy use?

cosine, dotproduct and euclidean. Cosine is the default and compares direction; dot product incorporates direction and magnitude; Euclidean distance measures straight-line distance. Choose the function expected by the embedding model, and use it consistently for indexing and search.

m-19-5-1 · Åpne dette flashcardet

How should a Cosmos DB document represent an embedding?

As a JSON array of numbers, for example "embedding": [0.12, -0.34, 0.56]. The container’s vector embedding policy identifies that property path and sets the numeric data type, dimension count and distance function. Each stored embedding and query vector must match that configuration.

m-24-1-1 · Åpne dette flashcardet

Does a Cosmos DB composite index accelerate VectorDistance queries?

No. Use a vector index, such as flat, quantizedFlat or DiskANN, for vector-distance search. A composite index combines ordinary property paths for filters/sorting; it does not perform nearest-neighbor vector search.

m-24-2-1 · Åpne dette flashcardet

Why exclude an embedding array from ordinary Cosmos DB property indexing?

An embedding array can contain thousands of values. Indexing each value as an ordinary property adds write work and index storage without providing vector similarity search. Exclude /embedding/* from ordinary indexing and configure the separate vector index on /embedding.

m-24-3-1 · Åpne dette flashcardet

What Cosmos DB SQL pattern returns the nearest five embeddings?

Use a query like: SELECT TOP 5 c.id, VectorDistance(c.embedding, @q) AS score FROM c ORDER BY VectorDistance(c.embedding, @q) Bind @q to the query’s numeric embedding array using the SDK’s query parameters. TOP 5 limits the result count; ORDER BY places the nearest distances first. The container needs the matching vector policy and an appropriate vector index.

m-24-5-1 · Åpne dette flashcardet

Where is a Cosmos DB vector embedding policy defined?

In the container configuration at creation time. It specifies the vector property path, data type, dimensions and distance function. Queries then provide a query vector and call VectorDistance to search that configured data.

# Send meldinger til Functions-hostens logg.

# Hent FunctionApp og dokumenttypene for Python-bindingene.
import azure.functions as func

# Opprett appen som Functions-hosten oppdager.
app = func.FunctionApp()


# Send endrede dokumenter til documents-argumentet.
# Databasenavnet hentes fra appinnstillingen COSMOS_DATABASE.
@app.cosmos_db_trigger(arg_name="documents", database_name="%COSMOS_DATABASE%",
                      # Overvåk notes med tilkoblingen fra CosmosConnection-innstillingen.
                      container_name="notes", connection="CosmosConnection",
                      # Lagre fremdrift i leases; opprett lease-containeren hvis den mangler.
                      lease_container_name="leases", create_lease_container_if_not_exists=True)
# En triggerkjøring kan inneholde en samling endrede dokumenter.
def changed(documents: func.DocumentList):
    # Behandle hvert dokument i leveransen.
    for document in documents:
        # Logg ID og revisjon slik at du kan sammenligne med oppdateringsskriptet.
        logging.info("changed id=%s revision=%s", document["id"], document.get("revision", 0))

What does Cosmos DB latest-version change feed omit?

Deletes and potentially intermediate updates. If an item changes several times before the reader catches up, it may receive only the latest version. The latest-version feed is therefore useful for reacting to current changes, but not as a complete audit log.

m-21-1-1 · Åpne dette flashcardet

What state does a Cosmos DB lease container store?

Ownership of feed ranges and checkpoints recording processing progress. Worker instances use these leases to divide work and resume after scaling or failure.

m-21-3-1 · Åpne dette flashcardet

Which partition key is used for a partitioned Cosmos DB lease container?

Use /id. Each lease document’s id is also its partition-key value. This is a requirement for a partitioned lease container used by the change feed processor; the monitored data container can use a different partition key.

m-21-4-1 · Åpne dette flashcardet

What are the four parts of a Cosmos DB change feed processor?

The monitored container holds changing items; the lease container stores ownership and progress; a compute instance runs the processor; and your callback handles each delivered batch.

m-25-1-1 · Åpne dette flashcardet

How do several Cosmos DB change feed processor instances divide work?

They share the same processor name and lease container, with a unique name for each instance. The processor assigns feed-range leases among instances and reassigns them as workers join or leave.

m-25-3-1 · Åpne dette flashcardet

Which Cosmos DB change-feed model invokes your handler with batches of changes?

The push model, through the change feed processor or a Functions Cosmos DB trigger. It reads incremental changes and calls your handler with batches. Your application does not repeatedly query every item in the container.

m-25-4-1 · Åpne dette flashcardet

How can Python consume Cosmos DB change feed without a native change feed processor client?

Use query_items_change_feed and save its continuation token to resume later. Alternatively, use a Python Azure Functions Cosmos DB trigger, letting the Functions host manage progress and invoke the Python handler. The native change feed processor clients are for .NET and Java.

What read guarantee does Cosmos DB Strong consistency provide?

The read returns the latest committed version of the item. This guarantee applies across the account’s regions. Strong consistency is unavailable when multiple write regions are enabled.

m-18-1-1 · Åpne dette flashcardet

How far behind can Cosmos DB reads be with Bounded staleness consistency?

At most the configured number of item versions, K, or time interval, T, whichever bound is reached first. Reads can be stale within those limits, but the service prevents replication lag from growing beyond them.

m-18-2-1 · Åpne dette flashcardet

Which Cosmos DB consistency level provides session read-your-own-writes?

Session consistency, the default. Reads using the same session token can see that session’s successful writes. Other client sessions do not automatically receive the same guarantee about those writes.

m-18-3-1 · Åpne dette flashcardet

What ordering guarantee does Cosmos DB Consistent prefix provide?

A reader sees writes in order, without gaps in the visible sequence. If writes happen as A, then B, then C, the reader might see only A or A and B, but not C followed by A. There is no configured maximum age for the data returned.

m-18-4-1 · Åpne dette flashcardet

What does Cosmos DB Eventual consistency guarantee?

If writes stop, replicas eventually converge on the same value. Until then, a read may return older data, with no configured maximum lag or guaranteed write ordering.

m-18-5-1 · Åpne dette flashcardet

What are the minimum Bounded staleness settings for single-region and multi-region Cosmos DB accounts?

Single-region accounts: 10 item versions or 5 seconds. Multi-region accounts: 100,000 versions or 300 seconds. These are the lowest permitted staleness-bound settings, not the normal lag every read will experience.

m-18-6-1 · Åpne dette flashcardet

Which Cosmos DB consistency levels bound cross-region staleness?

Strong and Bounded staleness. Strong returns the latest committed data; Bounded staleness permits lag only within configured version/time limits.

What distinguishes a Cosmos DB quantizedFlat vector index?

It compresses vectors to lower-precision values, then searches those values exhaustively. This can reduce index size and query cost, with a possible loss of accuracy. It supports up to 4,096 dimensions.

What distinguishes a Cosmos DB DiskANN vector index?

It uses a graph-based approximate search instead of comparing every candidate vector. This can make large searches more efficient, with a trade-off in exactness. It supports up to 4,096 dimensions.

What happens to Cosmos DB quantizedFlat or DiskANN searches with fewer than 1,000 vectors?

quantizedFlat and DiskANN searches fall back to a full scan. Those index types need at least 1,000 vectors for the intended quantized indexing behavior. Results are still possible, but the scan can cost more request units.

Within what range does Cosmos DB autoscale adjust provisioned RU/s?

From 10% to 100% of the configured maximum RU/s. For example, a maximum of 10,000 RU/s gives an autoscale range of 1,000–10,000 RU/s.

Which Cosmos DB indexing path wins when inclusion and exclusion overlap?

The more specific path. For example, including /* but excluding /largeText/? indexes other properties while omitting largeText. A deeper property path is more specific, and /? is more specific than /* at the same path. Avoid contradictory entries for an identical path.

How can indexing fewer Cosmos DB properties reduce write RU cost?

Each write updates fewer index entries, reducing the indexing work it consumes. Exclude fields the app does not need for indexed filters or sorting. For example, excluding /largeText/? avoids indexing that text while /* can keep other fields indexed.

What is the storage trade-off of float16 rather than float32 embeddings in Cosmos DB?

Each float16 component uses 16 bits instead of float32’s 32 bits, reducing raw vector size at the cost of numerical precision. Total document/index storage is not necessarily halved because it also includes other data and overhead.

Why must a Cosmos DB change-feed handler tolerate duplicate processing?

Because processing is at least once. A worker can finish its business action and fail before saving the checkpoint, causing that batch to be read again. Make the handler safe to repeat, for example by checking a stable operation ID before applying the same change twice.

How do Cosmos DB provisioned throughput and serverless differ?

Provisioned throughput reserves a capacity budget in RU/s, either fixed or autoscaled. Serverless charges for the request units actually consumed rather than continuously reserving that capacity. Both also charge for storage.

Which Cosmos DB feature automatically expires items after a configured lifetime?

Time to Live (TTL). Enable it on the container and set a default lifetime or an item-level ttl value. The lifetime runs from the item’s last modification, so an update restarts its clock.

A composite index covering the ORDER BY property sequence and compatible sort directions. For example, ORDER BY c.lastName ASC, c.firstName ASC needs a composite index for those paths together. Separate single-property indexes do not replace the required composite index for this ordering.
