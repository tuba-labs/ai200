"""Small isolated Cosmos DB lab; credentials are read from the environment."""
# Velg hvilken Cosmos-operasjon eksemplet skal utføre.
import argparse
# Les endpoint, nøkkel og databasenavn fra terminalens miljø.
import os

# Bruk Cosmos SDK-et og en eksplisitt partisjonsnøkkel.
from azure.cosmos import CosmosClient, PartitionKey

# Bruk små undervisningsdokumenter med håndskrevne vektorer.
DOCUMENTS = [
    # Første dokument ligger i partisjonen containers.
    {"id": "note-1", "topic": "containers", "text": "Package an application", "embedding": [1.0, 0.0, 0.0]},
    # Andre dokument ligger i samme partisjon, med en lignende vektor.
    {"id": "note-2", "topic": "containers", "text": "Inspect container logs", "embedding": [0.9, 0.1, 0.0]},
    # Tredje dokument viser hvordan metadatafilter kan utelate et nært vektortreff.
    {"id": "note-3", "topic": "data", "text": "Retrieve stored data", "embedding": [0.0, 1.0, 0.0]},
]


# La SDK-et sende responsheaderne til denne callbacken.
def report(headers, _body):
    # Hent kostnaden i request units fra responsheaderen.
    charge = headers.get("x-ms-request-charge")
    # Skriv bare RU når headeren finnes.
    if charge:
        # Vis grunnlaget for sammenligning av oppslag.
        print("Request charge:", charge, "RU")


# Kjør operasjonen som ble valgt på kommandolinjen.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Begrens valgene til operasjonene workshopene bruker.
    parser.add_argument("mode", choices=["setup", "query", "point", "vector-setup", "vector", "update"])
    # Les valgt operasjon.
    args = parser.parse_args()
    # Åpne klienten med labkontoens endpoint og nøkkel, og lukk den etter bruk.
    with CosmosClient(os.environ["COSMOS_ENDPOINT"], credential=os.environ["COSMOS_KEY"]) as client:
        # Use a database without shared throughput so vector search is supported.
        # Velg databasen; dette oppretter ikke databasen.
        database = client.get_database_client(os.getenv("COSMOS_DATABASE", "workshop"))
        # Hold vektorpolicyen i en egen container.
        name = "vectors" if args.mode.startswith("vector") else "notes"
        # Hent en klient for den valgte containeren.
        container = database.get_container_client(name)
        # Opprett labcontainer og eksempeldata ved setup.
        if args.mode in {"setup", "vector-setup"}:
            # Beskriv den vanlige indekseringspolicyen.
            policy = {
                # Oppdater indeksene automatisk når dokumentene endres.
                "indexingMode": "consistent", "automatic": True,
                # Indekser dokumentfeltene som standard.
                "includedPaths": [{"path": "/*"}],
                # Unnta vektorarrayet fra vanlig feltindeksering.
                "excludedPaths": [{"path": "/embedding/*"}],
            }
            # Legg bare til vektorinnstillinger for vektorcontaineren.
            extra = {}
            # Konfigurer en egen indeks for VectorDistance-søk.
            if args.mode == "vector-setup":
                # Bruk en flat vektorindeks på embedding-feltet.
                policy["vectorIndexes"] = [{"path": "/embedding", "type": "flat"}]
                # Beskriv hvilke vektorer embedding-feltet inneholder.
                extra["vector_embedding_policy"] = {"vectorEmbeddings": [{
                    # Lagre vektorene i embedding og behandle verdiene som float32.
                    "path": "/embedding", "dataType": "float32",
                    # Krev tre dimensjoner og sammenlign med cosinusavstand.
                    "dimensions": 3, "distanceFunction": "cosine",
                }]}
            # Opprett containeren hvis den mangler; eksisterende policy erstattes ikke her.
            container = database.create_container_if_not_exists(
                # Bruk topic som partisjonsnøkkel.
                id=name, partition_key=PartitionKey(path="/topic"),
                # Bruk policyen over og 400 RU/s dedikert throughput.
                indexing_policy=policy, offer_throughput=400, **extra,
            )
            # Skriv inn undervisningsdokumentene.
            for document in DOCUMENTS:
                # Opprett eller erstatt dokumentet, og vis operasjonens RU.
                container.upsert_item(document, response_hook=report)
            # Vis hvilken container som er klargjort.
            print("Ready:", name)
        # Utfør et direkte oppslag uten SQL-spørring.
        elif args.mode == "point":
            # ID og partisjonsnøkkel identifiserer dokumentet sammen.
            result = container.read_item("note-1", partition_key="containers", response_hook=report)
            # Vis dokumentets ID og tekst.
            print(result["id"], result["text"])
        # Lag en endring som change feed-triggeren kan observere.
        elif args.mode == "update":
            # Les dokumentet før oppdateringen.
            document = container.read_item("note-1", partition_key="containers")
            # Øk et synlig revisjonsnummer.
            document["revision"] = document.get("revision", 0) + 1
            # Lagre den nye versjonen og vis skrivekostnaden.
            container.upsert_item(document, response_hook=report)
            # Bruk revisjonen til å kjenne igjen endringen i triggerloggen.
            print("Updated note-1, revision", document["revision"])
        # Kjør vanlig SQL-søk eller vektorsøk.
        else:
            # Start med et metadatafilter.
            sql = "SELECT c.id, c.topic FROM c WHERE c.topic = @topic"
            # Send filterverdien som en parameter.
            parameters = [{"name": "@topic", "value": "containers"}]
            # Bytt til rangering etter vektoravstand.
            if args.mode == "vector":
                # Returner de to nærmeste treffene og deres avstand.
                sql = ("SELECT TOP 2 c.id, VectorDistance(c.embedding, @vector) AS distance "
                       # Behold emnefilteret og sorter de gjenværende dokumentene etter avstand.
                       "FROM c WHERE c.topic = @topic ORDER BY VectorDistance(c.embedding, @vector)")
                # Bruk samme tredimensjonale spørsmålsvektor som i PostgreSQL-øvelsen.
                parameters.append({"name": "@vector", "value": [0.0, 1.0, 0.0]})
            # Kjør den parameteriserte spørringen.
            results = container.query_items(sql, parameters=parameters,
                                            # Avgrens lesingen til containers-partisjonen og vis RU.
                                            partition_key="containers", response_hook=report)
            # Iterer resultatene fra SDK-et.
            for result in results:
                # Vis hvert treff i terminalen.
                print(result)


# Kjør bare når filen startes direkte.
if __name__ == "__main__":
    # Start den valgte laboperasjonen.
    main()

How many request units does a typical 1-KB Cosmos DB point read cost?

A typical 1-KB point read by id and partition key costs about 1 RU with Session, Consistent prefix or Eventual consistency. Strong and Bounded staleness reads cost about twice as much. A query is a different operation and can have a different cost.

How do you create a Cosmos DB client in Python?

Create one reusable CosmosClient with the account URL and an authorized credential: from azure.cosmos import CosmosClient from azure.identity import DefaultAzureCredential client = CosmosClient(endpoint, credential=DefaultAzureCredential()) Here endpoint is the account URL. The identity needs a Cosmos DB data-plane role; permission to manage the Azure resource alone is insufficient.

Which Cosmos DB SDK object performs item CRUD and queries?

Use the container client, called ContainerProxy in Python. For example, database.get_container_client("items") returns the object used for create_item, read_item, upsert_item, delete_item and query_items.

What is the Python SDK sequence for querying an existing Cosmos DB container?

Create or reuse the account client, select the existing database and container, then run the query: database = client.get_database_client("catalog") container = database.get_container_client("items") for item in container.query_items(query="SELECT * FROM c", enable_cross_partition_query=True): print(item) Here client is an authenticated CosmosClient. Replace the names with your existing resources. Selecting these clients does not create the database or container.

Should Python Cosmos DB code construct DatabaseProxy and ContainerProxy directly?

No. Ask the parent clients for them: database = client.get_database_client("catalog") container = database.get_container_client("items") The parent supplies the connection and configuration. Reuse these clients rather than manually constructing internal proxy objects.

m-23-5-1 · Åpne dette flashcardet

What two goals should a Cosmos DB partition key balance?

Balance distribution with access patterns. A useful key spreads data and writes across many values, while allowing common requests to target a known logical partition. For example, customerId can work when traffic is spread across customers. A single extremely busy customer can still create a hot partition.

prep-cosmos-006 · Åpne dette flashcardet

What happens when a Cosmos DB query cannot be routed to one partition?

The query can fan out across physical partitions and combine the results. That usually adds coordination and can increase RU consumption. Supplying a known partition key can narrow routing, but actual cost still depends on the query, indexes and data.

What distinguishes a Cosmos DB flat vector index?

A flat index performs exhaustive comparisons to find the exact nearest vectors under the selected metric. It offers full recall but supports at most 505 dimensions in Cosmos DB. Larger datasets can make exhaustive search more expensive.

m-19-1-1 · Åpne dette flashcardet

Which distance functions can a Cosmos DB vector embedding policy use?

The policy supports cosine, dotproduct and euclidean. Cosine compares direction; dot product also depends on magnitude; Euclidean distance measures straight-line separation. Use a metric compatible with the embedding model and keep stored vectors, queries and index settings consistent.

m-19-5-1 · Åpne dette flashcardet

How should a Cosmos DB document represent an embedding?

Store it as an array of numbers, for example "embedding": [0.12, -0.34, 0.56]. Configure that property in the container’s vector embedding policy. The array, query vector and policy must agree on dimensions, numeric representation and distance metric.

m-24-1-1 · Åpne dette flashcardet

Does a Cosmos DB composite index accelerate VectorDistance queries?

A composite index does not replace a vector index for nearest-neighbor search. Configure an appropriate flat, quantizedFlat or diskANN vector index for VectorDistance queries. Ordinary indexes can still help relevant scalar filters and sorting; their purpose differs from the vector index.

m-24-2-1 · Åpne dette flashcardet

Why exclude an embedding array from ordinary Cosmos DB property indexing?

Ordinary property indexing can index thousands of array elements without supplying vector similarity search. Excluding /embedding/* avoids that extra property-index work. Configure the separate vector index on /embedding for vector retrieval.

m-24-3-1 · Åpne dette flashcardet

What Cosmos DB SQL pattern returns the nearest five embeddings?

Use TOP to limit the result count and order by VectorDistance: SELECT TOP 5 c.id, VectorDistance(c.embedding, @q) AS score FROM c ORDER BY VectorDistance(c.embedding, @q) Bind @q to the query embedding through SDK query parameters. Stored vectors and the container’s vector policy must be compatible, with an appropriate vector index for efficient retrieval.

m-24-5-1 · Åpne dette flashcardet

Where is a Cosmos DB vector embedding policy defined?

Declare it in the container configuration at creation time. It identifies the vector property path, data type, dimension count and distance function. Later queries provide a compatible query vector to VectorDistance.

"""The Cosmos DB trigger uses the change feed processor and checkpoints in leases."""
# Send meldinger til Functions-hostens logg.
import logging
# Hent FunctionApp og dokumenttypene for Python-bindingene.
import azure.functions as func

# Opprett appen som Functions-hosten oppdager.
app = func.FunctionApp()


# Send endrede dokumenter til documents-argumentet.
# Databasenavnet hentes fra appinnstillingen COSMOS_DATABASE.
@app.cosmos_db_trigger(arg_name="documents", database_name="%COSMOS_DATABASE%",
                      # Overvåk notes med tilkoblingen fra CosmosConnection-innstillingen.
                      container_name="notes", connection="CosmosConnection",
                      # Lagre fremdrift i leases; opprett lease-containeren hvis den mangler.
                      lease_container_name="leases", create_lease_container_if_not_exists=True)
# En triggerkjøring kan inneholde en samling endrede dokumenter.
def changed(documents: func.DocumentList):
    # Behandle hvert dokument i leveransen.
    for document in documents:
        # Logg ID og revisjon slik at du kan sammenligne med oppdateringsskriptet.
        logging.info("changed id=%s revision=%s", document["id"], document.get("revision", 0))

What does Cosmos DB latest-version change feed omit?

It omits deletes and can omit intermediate versions when an item changes repeatedly before the reader catches up. It is useful for reacting to the latest item state, rather than reconstructing a complete audit history.

m-21-1-1 · Åpne dette flashcardet

What state does a Cosmos DB lease container store?

It stores leases identifying which worker owns each feed range, together with checkpoints recording how far processing has reached. This lets workers share work and resume after a restart or ownership change.

m-21-3-1 · Åpne dette flashcardet

Which partition key is used for a partitioned Cosmos DB lease container?

Use /id for a partitioned lease container used by the change feed processor. Each lease document’s id supplies that partition-key value. The monitored data container has its own independently chosen partition key.

m-21-4-1 · Åpne dette flashcardet

What are the four parts of a Cosmos DB change feed processor?

The monitored container holds the changing items. The lease container stores ownership and progress. Compute instances run the processor, and your callback handles each delivered batch. Together these components support distributed, resumable processing.

m-25-1-1 · Åpne dette flashcardet

How do Cosmos DB change feed processor and instance names control shared versus independent consumption?

The processor name identifies a processing group; an instance name identifies one worker in that group. Shared work: use the same source container, lease container and processor name, but unique instance names (worker-1, worker-2). Workers split the feed-range leases, so each handles a subset. Independent full feeds: use different processor names (search-index, audit), or separate lease containers. Each group has its own leases and saved reading positions. To give every worker an independent full feed, put each worker in a separate group. Different instance names alone still divide the work. Each group reads from its saved or configured starting position, within the selected feed mode. At-least-once delivery means changes can be processed again.

m-25-3-1 · Åpne dette flashcardet

Which Cosmos DB change-feed model invokes your handler with batches of changes?

The push model. The change feed processor polls Cosmos DB in the background, then automatically invokes your callback (the function you provide) with batches of changes. Push describes how the processor delivers batches to your code; it still polls Cosmos DB to retrieve them. The processor manages polling, work distribution and checkpoints (saved reading positions). In the pull model, your code explicitly requests batches and saves its own reading position.

m-25-4-1 · Åpne dette flashcardet

How can Python consume Cosmos DB change feed without a native change feed processor client?

Use Python’s query_items_change_feed and save its continuation token for later reads. Your code manages polling and progress. Alternatively, use a Python Azure Functions Cosmos DB trigger, which invokes your handler while the host manages change-feed processing. The native processor libraries documented here are for .NET and Java.

What read guarantee does Cosmos DB Strong consistency provide?

Strong consistency returns the latest committed item version. It provides a stronger freshness guarantee across the account’s regions, with corresponding latency and availability trade-offs. Cosmos DB does not support this level with multiple write regions.

m-18-1-1 · Åpne dette flashcardet

How far behind can Cosmos DB reads be with Bounded staleness consistency?

Lag is bounded by the configured version count K or time interval T, whichever limit is reached first. The service constrains cross-region replication lag; exceeding the bound can cause writes for an affected partition to be throttled until replication catches up.

m-18-2-1 · Åpne dette flashcardet

Which Cosmos DB consistency level provides session read-your-own-writes?

Session consistency provides read-your-own-writes when reads carry the relevant session token. The SDK tracks these tokens for a reused client. If work moves to another client, preserve the session context when that guarantee is required.

m-18-3-1 · Åpne dette flashcardet

What does Cosmos DB Consistent prefix guarantee for transactional batch writes?

The current documentation describes consistent visibility for writes made together in a transactional batch: a reader sees the committed changes together rather than a mixture of versions from that transaction. It describes separate single-document writes at this level as eventually consistent. Avoid treating the level as a global ordering guarantee for arbitrary independent writes.

m-18-4-1 · Åpne dette flashcardet

What does Cosmos DB Eventual consistency guarantee?

Replicas converge if writes stop, but reads may return older data while replication catches up. Eventual consistency does not set a maximum lag or guarantee that successive reads will observe monotonically newer versions.
