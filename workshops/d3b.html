Prøv complete, abandon, defer og dead-letter

  1. Send fire meldinger med ulike message-id-er, én om gangen. Motta den første med complete og den neste med abandon. Observer delivery count ved neste mottak.
  2. Defer en melding og noter sequence number fra skriptet. Hent den med --sequence og complete. Dead-letter en annen og les DLQ uten å fullføre før årsaken er notert.
  3. Gjenta abandon til max delivery count nås. Sammenlign peek med receive og vanlig kø med DLQ. Ingen av disse operasjonene gjør SQL-oppdateringen i et annet system atomisk.

"""Explicit Service Bus settlement; peek and DLQ inspection do not remove messages."""
# Velg en avgrenset operasjon på en labkø.
import argparse
# Les namespace fra terminalen.
import os


# Kjør én melding gjennom en synlig tilstandsendring.
def main():
    # Beskriv kommandolinjen.
    parser = argparse.ArgumentParser()
    # DLQ og peek skal bare inspisere.
    parser.add_argument("mode", choices=["send", "peek", "receive", "dlq"])
    # Bruk en egen labkø, med mulighet for session-kø i tillegg.
    parser.add_argument("--queue", default="documents")
    # Stabil id gjør det mulig å følge meldingen i notatene.
    parser.add_argument("--id", default="workshop-message")
    # Angi en session bare for en session-aktivert kø.
    parser.add_argument("--session")
    # Velg hvordan en faktisk mottatt melding gjøres opp.
    parser.add_argument("--settle", choices=["complete", "abandon", "defer", "dead-letter"], default="complete")
    # Deferred meldinger hentes eksplisitt med dette sekvensnummeret.
    parser.add_argument("--sequence", type=int)
    # Les valgene før nettverk brukes.
    args = parser.parse_args()
    # Bruk Entra-credential og Service Bus sine dataplanklienter.
    from azure.identity import DefaultAzureCredential
    # Hent riktig message- og subqueue-type.
    from azure.servicebus import ServiceBusClient, ServiceBusMessage, ServiceBusSubQueue
    # Lukk nettverksklient og credential etter den valgte operasjonen.
    with DefaultAzureCredential() as credential, ServiceBusClient(os.environ["SERVICEBUS_NAMESPACE"], credential) as client:
        # Sending oppretter en melding, ikke en mottakerlås.
        if args.mode == "send":
            # Velg køen fra kommandolinjen.
            with client.get_queue_sender(args.queue) as sender:
                # Bruk bare ufarlig undervisningsinnhold.
                sender.send_messages(ServiceBusMessage("Workshop payload", message_id=args.id, session_id=args.session))
            # Bekreft meldings-ID-en.
            print("Sent", args.id)
            # Avslutt etter sending.
            return
        # Legg bare på session-valget når det er oppgitt.
        options = {"session_id": args.session} if args.session else {}
        # Velg DLQ for inspeksjon uten å tømme den.
        if args.mode == "dlq":
            # Dette er køens innebygde dead-letter-underkø.
            options["sub_queue"] = ServiceBusSubQueue.DEAD_LETTER
        # Opprett en mottaker i den valgte køen.
        with client.get_queue_receiver(args.queue, max_wait_time=5, **options) as receiver:
            # Peek gir inspeksjon uten settlement.
            if args.mode in {"peek", "dlq"}:
                # Hent bare noen få metadataobjekter.
                messages = receiver.peek_messages(max_message_count=10)
            # En deferred melding ligger utenfor vanlig mottaksflyt.
            elif args.sequence is not None:
                # Hent akkurat det noterte sekvensnummeret.
                messages = receiver.receive_deferred_messages(sequence_numbers=[args.sequence])
            # Vanlig mottak tar én melding med PeekLock.
            else:
                # Avgrens forsøket så hvert resultat er lett å følge.
                messages = receiver.receive_messages(max_message_count=1, max_wait_time=5)
            # Vis hvilke meldinger operasjonen faktisk fant.
            for message in messages:
                # Logg bare lab-ID og leveringsmetadata.
                print("id=", message.message_id, "sequence=", message.sequence_number,
                      # DLQ-årsak er nyttig ved inspeksjon.
                      "delivery=", message.delivery_count, "reason=", message.dead_letter_reason)
                # Peek-objekter skal ikke fullføres eller endres.
                if args.mode in {"peek", "dlq"}:
                    # Fortsett uten settlement.
                    continue
                # Fullfør en vellykket behandlet melding.
                if args.settle == "complete":
                    # Fjern meldingen fra mottakskøen.
                    receiver.complete_message(message)
                # Returner en låst melding til leveringsflyten.
                elif args.settle == "abandon":
                    # Den kan mottas på nytt.
                    receiver.abandon_message(message)
                # Sett meldingen til side med sekvensnummeret bevart.
                elif args.settle == "defer":
                    # Neste vanlige mottak hopper over den.
                    receiver.defer_message(message)
                # Flytt en kontrollert feil til DLQ med synlig årsak.
                else:
                    # Dette er en labhandling, ikke standardfeilhåndtering for alle feil.
                    receiver.dead_letter_message(message, reason="Workshop", error_description="Controlled settlement experiment")


# Unngå Azure-kall når modulen importeres.
if __name__ == "__main__":
    # Kjør valgt meldingstilstandsovergang.
    main()

What does completing a Service Bus message do?

Completing a message tells Service Bus that processing succeeded and removes it from the queue or subscription. In PeekLock mode, receiving the message only reserves it temporarily; completion is the separate acknowledgment. Complete after the business operation succeeds. Make that operation safe to repeat in case it succeeds but the acknowledgment fails.

Eksempel · Python

receiver.complete_message(message)

With a live Python ServiceBusReceiver in PeekLock mode and its received message, call this after successful business processing. It acknowledges completion and removes the message.

What does abandoning a Service Bus message do?

Abandoning releases the lock and makes the message available for another delivery, normally increasing DeliveryCount. Use it when the current attempt cannot finish but another attempt may succeed. It does not delete the message or impose a retry delay. Another receiver may get it immediately.

Eksempel · Python

receiver.abandon_message(message)

With a live receiver and its locked message, this releases the message for another delivery attempt. It does not schedule a particular retry delay.

m-41-2-1 · Åpne dette flashcardet

How does Service Bus message deferral work?

Deferral sets the message aside so ordinary receives no longer return it. Save its SequenceNumber, then use the deferred-message receive operation to retrieve it later. The message remains in Service Bus. Deferring does not itself increase DeliveryCount, and your application must remember when and how to revisit it.

Eksempel · Python

sequence = message.sequence_number
receiver.defer_message(message)
messages = receiver.receive_deferred_messages(sequence_numbers=[sequence])

Save the sequence number before deferral. The last line represents a later retrieval when the application is ready; ordinary receives do not pick the deferred message up.

m-41-3-1 · Åpne dette flashcardet

What does dead-lettering a Service Bus message do?

It moves the message into the queue or subscription's dead-letter subqueue, where normal consumers no longer receive it. Record a reason so someone can diagnose the failure. An application or operator must later inspect, fix, resubmit or remove it.

Eksempel · Python

receiver.dead_letter_message(message, reason="InvalidPhotoUrl", error_description="Photo URL is missing")

On the receiver that owns this locked message, record a useful application-defined reason. The message moves to the entity’s dead-letter subqueue for explicit recovery.

m-41-4-1 · Åpne dette flashcardet

When does Service Bus automatically dead-letter a repeatedly delivered message?

When its delivery count exceeds the entity’s MaxDeliveryCount setting, which defaults to 10. Failed PeekLock deliveries followed by abandonment or lock expiry increase that count. Service Bus moves the message to the dead-letter queue with reason MaxDeliveryCountExceeded so it stops cycling through normal consumers.

m-42-2-1 · Åpne dette flashcardet

How are messages shared among consumers of a Service Bus queue?

They compete for the queue's messages. Each locked message is assigned to one receiver at a time, sharing the workload among consumers. A failed message can be delivered again. Use sessions if related messages must be processed in order.

Eksempel · Scenario

Two workers receive from the same repairs queue. A can lock repair-42 while B locks repair-43. A failed locked job can later be redelivered; creating B did not give it a separate copy of all work.

m-44-1-1 · Åpne dette flashcardet

Where are Service Bus dead-letter queues created?

Each queue and topic subscription has an automatically provided dead-letter subqueue. You do not create it as a separate top-level queue. Receive from that subqueue to inspect failed messages, then deliberately complete, repair or resubmit them. There is no automatic cleanup just because their original lifetime expires.

Eksempel · Service Bus entity paths

repairs/$DeadLetterQueue
orders/Subscriptions/billing/$DeadLetterQueue

The first path is the repairs queue’s dead-letter subqueue. The second belongs to billing’s subscription on topic orders. These subqueues are provided automatically rather than created as separate queues.

m-44-4-1 · Åpne dette flashcardet

Which Service Bus operation inspects a message without receiving and settling it?

Peek. It inspects stored messages without locking or removing them. It is useful for diagnostics but does not reserve a message for your worker; another receiver may process it. Use a normal receive operation when you need to own and settle work.

Eksempel · Python

messages = receiver.peek_messages(max_message_count=5)

This inspects up to five stored messages using an existing receiver. It does not lock them for processing or settle them.

prep-sb-010 · Åpne dette flashcardet

Where is a binary Service Bus payload stored?

The payload is in the message body. For example, serialize an order as JSON, encode it as bytes and send those bytes as the body. The receiver decodes them using the agreed format. Use application properties for supported routing metadata. Service Bus does not interpret the business meaning of the body.

import argparse
# Les Service Bus-navnet fra miljøet.
import os

# Bruk Entra-identitet, for eksempel den innloggede Azure CLI-brukeren.
from azure.identity import DefaultAzureCredential
# Hent klienten, meldingstypen og valget for dead-letter-køen.
from azure.servicebus import ServiceBusClient, ServiceBusMessage, ServiceBusSubQueue


# Kjør én operasjon mot labnamespace-et.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Begrens operasjonene til dem som brukes i øvelsene.
    parser.add_argument("mode", choices=["send", "receive", "dlq", "topic", "subscription"])
    # Velg mottakende subscription; indexer er standard.
    parser.add_argument("--subscription", default="indexer")
    # Les valgene fra terminalen.
    args = parser.parse_args()
    # Opprett identitet og klient, og lukk dem etter operasjonen.
    with DefaultAzureCredential() as credential, ServiceBusClient(
            # Bruk namespace-adressen uten å lagre en tilkoblingsnøkkel i filen.
            fully_qualified_namespace=os.environ["SERVICEBUS_NAMESPACE"], credential=credential) as client:
        # Disse operasjonene sender én ny melding.
        if args.mode in {"send", "topic"}:
            # Send til topic-et når topic-operasjonen er valgt.
            sender = (client.get_topic_sender("document-events") if args.mode == "topic"
                      # Ellers sendes meldingen til documents-køen.
                      else client.get_queue_sender("documents"))
            # Åpne senderen for denne sendingen.
            with sender:
                # Bruk en liten JSON-body som nyttelast.
                sender.send_messages(ServiceBusMessage('{"id":"doc-001"}',
                                                       # Legg til subject og en egenskap som subscription-filtre kan bruke.
                                                       subject="document", application_properties={"kind": "note"}))
            # Bekreft at SDK-sendingen er fullført.
            print("Sent")
        # De øvrige operasjonene mottar meldinger.
        else:
            # En topic-mottaker leser fra en bestemt subscription.
            if args.mode == "subscription":
                # Velg indexer eller audit på document-events-topic-et.
                receiver = client.get_subscription_receiver("document-events", args.subscription)
            # Køøvelsen leser enten vanlig kø eller dens DLQ.
            else:
                # Opprett mottakeren for documents-køen.
                receiver = client.get_queue_receiver("documents", sub_queue=(
                    # Velg dead-letter-underkøen bare i dlq-modus.
                    ServiceBusSubQueue.DEAD_LETTER if args.mode == "dlq" else None))
            # Åpne mottakeren og lukk den etter bruk.
            with receiver:
                # Hent høyst ti meldinger og begrens ventingen til fem sekunder.
                messages = receiver.receive_messages(max_message_count=10, max_wait_time=5)
                # Vis hvor mange meldinger som ble hentet.
                print("Received:", len(messages))
                # Behandle hver mottatt melding.
                for message in messages:
                    # Vis body og antall leveringsforsøk.
                    print(str(message), "Delivery:", message.delivery_count)
                    # Fremprovoser DLQ-flyten i den vanlige køøvelsen.
                    if args.mode == "receive":
                        # Flytt meldingen til DLQ med en årsak du kan inspisere.
                        receiver.dead_letter_message(message, reason="Workshop", error_description="Controlled demo")
                    # Fullfør meldinger i DLQ- og subscription-øvelsene.
                    else:
                        # Bekreft behandling og fjern meldingen fra denne mottakerkøen.
                        receiver.complete_message(message)

Which Service Bus entities give independent consumers their own copies?

A topic with a separate subscription for each independent consumer. For example, billing and shipping subscriptions each receive a copy of matching order messages. Several workers reading the same subscription compete for its messages, just as workers on one queue do.

m-44-2-1 · Åpne dette flashcardet

What do Service Bus subscription rules filter?

Subscription rules select topic messages by supported system and application properties. SQL filters evaluate expressions; correlation filters match specified values. Neither inspects the message body. For example, publish an order with application properties region="NO" and priority=8. Give the subscription this SQL filter: region = 'NO' AND priority >= 5 The order matches. An order with region="SE" or priority=2 does not match. A correlation filter configured only with region="NO" matches Norwegian orders regardless of priority. These must be message properties. Putting region and priority only in a JSON body does not make them filterable. Remove the default $Default accept-all rule when adding restrictive rules. Rules are alternatives: any matching rule admits a message. If $Default remains, both Norwegian and Swedish orders are admitted.

m-44-3-1 · Åpne dette flashcardet

Which Service Bus tiers support topics and subscriptions?

Standard and Premium. Basic supports queues but does not support topics and subscriptions.

Azure Service Bus Data Receiver. Assign it at the narrowest appropriate namespace/entity scope to let an identity receive and settle messages. It does not grant send permission. Authenticate with Microsoft Entra ID, then let the data role determine the permitted operations.

prep-p-sb-009 · Åpne dette flashcardet

Which Service Bus data role permits sending without receiving?

Azure Service Bus Data Sender. It permits publishing messages at the assigned scope without granting receive permission. Use a managed identity or another Microsoft Entra principal and assign the role to the queue/topic or suitable parent scope.

prep-p-rbac-024 · Åpne dette flashcardet

How does a Service Bus correlation filter differ from a SQL subscription filter?

A correlation filter directly matches specified property values, such as CorrelationId or an application property. A SQL filter evaluates an expression, such as priority > 5 AND region = 'NO'. Correlation filters suit simple exact matches and generally cost less broker processing. Neither parses arbitrary message-body content.

What does Service Bus auto-forwarding do?

Auto-forwarding transfers messages from a queue or subscription into a configured queue or topic in the same namespace. Successful transfer removes them from the source. Use subscriptions when independent consumers need their own copies; forwarding creates a delivery chain.

Eksempel · Forwarding flow

incoming queue → processing queue

Both entities are in the same Service Bus namespace. Successful forwarding moves the message onward; it is not independent fan-out to two consumer subscriptions.

m-44-5-1 · Åpne dette flashcardet

When does a failed Service Bus PeekLock attempt increase DeliveryCount?

Abandoning a message or letting its lock expire normally increases DeliveryCount and allows redelivery. Some infrastructure-related lock losses do not increase it. The MessageLockLost exception alone does not tell you which case occurred.

m-41-5-1 · Åpne dette flashcardet

What are Service Bus PeekLock's default and maximum lock durations?

The default entity lock duration is one minute, and a single lock period can be configured up to five minutes. A receiver can renew the lock to continue processing beyond that period. Renewal is not guaranteed forever: handle failures and make the work safe to repeat if the message is redelivered.

m-42-1-1 · Åpne dette flashcardet

How does Service Bus duplicate detection identify a duplicate message?

With duplicate detection enabled, Service Bus remembers MessageId values within a configured window and discards repeat sends with the same identity. The window defaults to ten minutes and supports 20 seconds to seven days. On partitioned entities, MessageId plus PartitionKey defines uniqueness. Enable the feature when creating the queue/topic; it does not prevent redelivery of an already accepted message after a consumer failure.

Eksempel · Scenario

A sender retries the same business send with MessageId repair-42 inside the configured detection window. The broker can discard that repeat send. The already accepted message can still be redelivered if a receiver later fails.

m-42-3-1 · Åpne dette flashcardet

How can Service Bus group related messages for ordered processing?

Enable sessions and give related messages the same SessionId. A receiver obtains exclusive ownership of that session and can process its messages in first-in, first-out order. Different sessions can be processed in parallel. The handler must preserve that order too: starting several operations concurrently or abandoning a message can change the order in which business work finishes.

Eksempel · Scenario

Created, assigned and completed messages for repair 42 share SessionId repair-42 and are enqueued in that order. One session receiver can handle them sequentially while a different session receiver handles repair 43.

m-42-4-1 · Åpne dette flashcardet

How does Service Bus PeekLock differ from ReceiveAndDelete?

PeekLock temporarily locks the message but keeps it in Service Bus until you complete or otherwise settle it. A failed attempt can be redelivered. ReceiveAndDelete removes it as part of receipt, so a crash before processing finishes can lose the work. Choose based on whether retries and explicit acknowledgment are required.

prep-sb-002 · Åpne dette flashcardet

How is a Service Bus message scheduled for future availability?

Use the SDK’s scheduling operation with a future UTC time. Service Bus keeps the message unavailable until that time, then makes it available for consumers. This schedules availability, rather than an exact execution time: a busy or offline consumer can process it later.

Eksempel · Python

from datetime import datetime, timedelta, timezone
sender.schedule_messages(message, datetime.now(timezone.utc) + timedelta(minutes=10))

With an existing Python ServiceBusSender and message, this schedules availability ten minutes from now. Actual processing also depends on consumers being ready.

How does a transactional outbox coordinate a database update with message publication?

Write the business change and an “event to publish” record in the same database transaction. A separate relay reads that outbox and sends the event to Service Bus, retrying as needed. This avoids committing the business change while forgetting its event. The relay may send twice, so use stable IDs and duplicate-safe consumers; it is not one transaction across the database and broker.

Eksempel · Scenario

One database transaction closes repair 42 and adds an outbox record for RepairCompleted. If the broker is temporarily unavailable, the relay can retry the saved record later instead of losing the event after the repair update committed.

What happens when a Service Bus message expires with dead-lettering on expiration enabled?

For ordinary active messages, Service Bus moves the expired message to the dead-letter queue with reason TTLExpiredException. It cannot be processed through normal delivery afterward. Deferred messages are an exception: expiration does not automatically sweep them into the dead-letter queue. Monitor the dead-letter queue and handle deferred work explicitly.

How can a Service Bus handler keep a PeekLock during long processing?

Renew the lock before it expires. You can renew it explicitly or configure the processor's automatic lock renewal for the expected processing time. Renewals can fail, so handle lost locks and make repeated processing safe.

Eksempel · Python

receiver.renew_message_lock(message)

Renew a message lock before it expires while using the receiver that owns it. A renewal can fail, so completion/retry handling must still be safe.

How can a Service Bus Functions binding connect without a shared-access connection string?

Configure an identity-based binding connection, for example Orders__fullyQualifiedNamespace=mynamespace.servicebus.windows.net, and set the binding’s connection name to Orders. Give the Function App’s managed identity the required data role, commonly Azure Service Bus Data Receiver for a trigger. The binding then obtains tokens for that identity instead of using a shared-access connection string.

What does the claim-check messaging pattern put in the broker message?

A reference to the payload. Store the large content in durable storage, such as Blob Storage, and send its identifier or URL in the message. The receiver fetches the content using that reference. Keep it accessible until the message no longer needs it.

Eksempel · JSON · example message body

{"repairId":"42","payloadUrl":"https://studystore.blob.core.windows.net/photos/42.jpg"}

The broker carries this small reference. The consumer retrieves the large photo using its own authorized storage access; keep the payload available for the message’s processing/recovery lifetime.

Which Service Bus metric helps identify a growing backlog of unprocessed work?

ActiveMessages measures active queue depth: messages available for consumers to process. A sustained rise means pending work is accumulating faster than consumers remove it. Compare it with send/receive rates, processing duration and failures. Queue depth shows the symptom; these other measurements help distinguish slow workers, errors and a traffic surge.

Eksempel · Illustrative queue-depth measurements

10:00  100 active messages
10:05  250 active messages
10:10  500 active messages

A sustained rise indicates accumulating pending work. Compare send/receive rates, processing time and failures before deciding whether to add consumers or fix a processing problem.

prep-p-sb-019 · Åpne dette flashcardet

What does Service Bus AutoDeleteOnIdle configure?

Automatic deletion of a supported queue, topic or subscription after its configured period of inactivity. The minimum interval is five minutes. Service Bus defines activity differently for different entity types, so do not assume it means only time since the last send. It is useful for deliberately temporary messaging entities.

prep-p-sb-021 · Åpne dette flashcardet

What can a Service Bus subscription rule action modify?

A SQL rule action can set, remove or change supported message properties on the copy delivered through that rule. It cannot run arbitrary application code or rewrite the binary body. Filters choose matching messages; actions adjust their properties for that subscription.

Eksempel · Service Bus SQL rule action

SET user.route = 'photo-processing'

This action sets an application property on the copy admitted through the rule. It does not rewrite the message’s binary body or execute your application handler.

prep-sb-030 · Åpne dette flashcardet

Which Service Bus properties coordinate a request/reply exchange?

ReplyTo identifies the queue or topic to which the receiver should send a response. CorrelationId links that response to the original request, often by copying the request’s MessageId. The applications must implement this convention. Setting ReplyTo does not cause Service Bus to generate a reply or send one automatically.

Eksempel · Request/reply property examples

Request: MessageId=request-42, ReplyTo=reply-orders
Reply:   CorrelationId=request-42

The receiver application sends a reply to reply-orders and copies the original identifier for correlation. Service Bus does not generate this application response automatically.

prep-sb-028 · Åpne dette flashcardet

What can a Service Bus broker transaction combine?

Supported sends and message-settlement operations within Service Bus. For example, completing an input message and sending a follow-up can commit together: both happen or neither does. The tier, SDK and cross-entity settings must support the combination. Updating an external database requires a separate coordination pattern, such as a transactional outbox.

"""Send ordered teaching events and receive using durable Blob checkpoints."""
# Velg operasjon og consumer group.
import argparse
# Lag en liten JSON-body for hver hendelse.
import json
# Les navn, endpoint og checkpoint-lager fra miljøet.
import os


# Bygg kun ufarlige og forutsigbare undervisningshendelser.
def teaching_events():
    # Bruk samme dokumentemne for alle fem hendelser.
    return [{"id": f"document-{number}", "topic": "containers", "revision": number} for number in range(1, 6)]


# Kjør den valgte nettverksoperasjonen.
def main():
    # Beskriv kommandolinjen før SDK-et lastes.
    parser = argparse.ArgumentParser()
    # Tillat sending eller mottak.
    parser.add_argument("mode", choices=["send", "receive"])
    # Egen gruppe holder leseposisjonen separat fra andre apper.
    parser.add_argument("--group", default="ai200-a")
    # Les valgene.
    args = parser.parse_args()
    # Bruk synkrone Event Hubs-klienter.
    from azure.eventhub import EventData, EventHubConsumerClient, EventHubProducerClient
    # Lagre fremdrift i et varig Blob-lager.
    from azure.eventhub.extensions.checkpointstoreblob import BlobCheckpointStore
    # Bruk lokal CLI-identitet eller støttet Azure-identitet.
    from azure.identity import DefaultAzureCredential
    # Del de ikke-hemmelige klientinnstillingene.
    settings = dict(fully_qualified_namespace=os.environ["EVENTHUB_NAMESPACE"], eventhub_name=os.environ["EVENTHUB_NAME"])
    # Lukk credential etter at operasjonen er ferdig.
    with DefaultAzureCredential() as credential:
        # Send fem hendelser i en batch til én valgt partisjon.
        if args.mode == "send":
            # Åpne produsenten.
            with EventHubProducerClient(credential=credential, **settings) as producer:
                # Samme partisjonsnøkkel holder relaterte hendelser sammen.
                batch = producer.create_batch(partition_key="containers")
                # Legg inn hendelsene i original rekkefølge.
                for item in teaching_events():
                    # SDK-et kontrollerer batchstørrelsen ved add.
                    batch.add(EventData(json.dumps(item)))
                # Send batchen til tjenesten.
                producer.send_batch(batch)
            # Bekreft sending uten å skrive tilkoblingsdata.
            print("Sent 5 events with partition key containers")
            # Avslutt senderoperasjonen.
            return
        # Opprett varig eierskaps- og checkpoint-lager for konsumentgruppen.
        store = BlobCheckpointStore(blob_account_url=os.environ["CHECKPOINT_ACCOUNT_URL"],
                                    # Bruk en labcontainer som allerede er opprettet.
                                    container_name=os.getenv("CHECKPOINT_CONTAINER", "checkpoints"), credential=credential)
        # Kall handleren etter mottak av en hendelse.
        def on_event(context, event):
            # max_wait_time kan gi en tom callback; den skal ikke checkpointes.
            if event is None:
                # Vent videre på faktisk data.
                return
            # Vis rekkefølge innen partisjonen og den ufarlige testbodyen.
            print(context.partition_id, event.sequence_number, event.body_as_str(), flush=True)
            # Bekreft bare etter at denne enkle handleren er ferdig; per-event er valgt for synlighet.
            context.update_checkpoint(event)
        # Åpne en konsument som deler eierskap bare innen valgt gruppe.
        with EventHubConsumerClient(consumer_group=args.group, checkpoint_store=store, credential=credential, **settings) as consumer:
            # La Ctrl+C være den eksplisitte avslutningen i terminaløvelsen.
            try:
                # Begynn ved tilgjengelig start bare når gruppen ikke allerede har et checkpoint.
                consumer.receive(on_event=on_event, starting_position="-1", max_wait_time=5)
            # Avslutt normalt og la with-blokken lukke klienten.
            except KeyboardInterrupt:
                # Bekreft at deltakeren stoppet mottaket.
                print("Stopped; processed events have checkpoints")


# Ingen Azure-operasjon skjer ved import.
if __name__ == "__main__":
    # Kjør den valgte senderen eller mottakeren.
    main()
