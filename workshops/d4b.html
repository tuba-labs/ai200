"""Two separate HTTP services with explicit W3C context propagation and metrics."""
# Velg frontend eller backend og eventuelle lokale porter.
import argparse
# Bygg HTTP-responsen med trace-ID som JSON.
import json
# Skriv rolle, trace-ID og status til konsollen.
import logging
# Les Azure Monitor-tilkoblingen fra miljøet.
import os
# Mål svartid og lag en kontrollert forsinkelse.
import time
# Importer HTTP-feiltypene som hører til nettverksklienten.
import urllib.error
# La frontend gjøre et faktisk HTTP-kall til backend.
import urllib.request
# Kjør to små HTTP-servere i separate prosesser.
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
# Les delay og fail fra URL-ens spørringsparametere.
from urllib.parse import parse_qs, urlsplit

# Hent API-ene for målinger, sporingskontekst og spans.
from opentelemetry import metrics, propagate, trace
# Konfigurer hvor målinger samles og eksporteres.
from opentelemetry.sdk.metrics import MeterProvider
# Eksporter aggregerte målinger med et fast intervall.
from opentelemetry.sdk.metrics.export import PeriodicExportingMetricReader
# Gi telemetrien et tjenestenavn som skiller prosessene.
from opentelemetry.sdk.resources import Resource
# Konfigurer sporings-SDK-et.
from opentelemetry.sdk.trace import TracerProvider
# Velg købasert Azure-eksport eller direkte konsollutskrift.
from opentelemetry.sdk.trace.export import BatchSpanProcessor, ConsoleSpanExporter, SimpleSpanProcessor
# Velg en eksplisitt sampler for det nye eksperimentet.
from opentelemetry.sdk.trace.sampling import ALWAYS_ON, ParentBased, TraceIdRatioBased
# Merk klient-/serverspans og feilstatus eksplisitt.
from opentelemetry.trace import SpanKind, Status, StatusCode


# Klargjør samme telemetrioppsett for hver av de to prosessene.
def setup(role, sampler_mode="all"):
    # Skill ai200-frontend fra ai200-backend i Azure og konsollen.
    resource = Resource.create({"service.name": "ai200-" + role})
    # Opprett sporingsleverandøren med tjenestens identitet.
    # En root-ratio kan brukes alene eller med barn som følger upstream-beslutningen.
    ratio = TraceIdRatioBased(0.1)
    # Standardvalget i denne laben er eksplisitt all, uavhengig av distro-standarder.
    sampler = ALWAYS_ON if sampler_mode == "all" else ParentBased(ratio) if sampler_mode == "parent-ratio" else ratio
    # Koble sampleren til leverandøren før spans opprettes.
    provider = TracerProvider(resource=resource, sampler=sampler)
    # Uten Azure-tilkobling opprettes ingen metrikkeksportør.
    readers = []
    # La miljøvariabelen velge om telemetrien skal sendes til Azure.
    connection = os.getenv("APPLICATIONINSIGHTS_CONNECTION_STRING")
    # Bruk Azure Monitor når en connection string er konfigurert.
    if connection:
        # Hent eksportørene for både målinger og spans.
        from azure.monitor.opentelemetry.exporter import AzureMonitorMetricExporter, AzureMonitorTraceExporter
        # Send spans i bakgrunnen i samlinger.
        provider.add_span_processor(BatchSpanProcessor(AzureMonitorTraceExporter(connection_string=connection)))
        # Koble en periodisk leser til måleeksportøren.
        readers.append(PeriodicExportingMetricReader(
            # Bruk samme Azure-ressurs og eksporter målinger hvert tiende sekund.
            AzureMonitorMetricExporter(connection_string=connection), export_interval_millis=10000))
    # Bruk konsollen når Azure-tilkoblingen mangler.
    else:
        # Skriv hver avsluttet span til konsollen for lokal inspeksjon.
        provider.add_span_processor(SimpleSpanProcessor(ConsoleSpanExporter()))
    # Gjør dette sporingsoppsettet aktivt i prosessen.
    trace.set_tracer_provider(provider)
    # La målingene bruke samme tjenestenavn som sporingen.
    meter_provider = MeterProvider(resource=resource, metric_readers=readers)
    # Gjør måleoppsettet aktivt i prosessen.
    metrics.set_meter_provider(meter_provider)
    # Behold leverandørene slik at de kan flushes ved avslutning.
    return provider, meter_provider


# Start én av tjenestene fra kommandolinjen.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Velg hvilken tjeneste denne prosessen skal være.
    parser.add_argument("role", choices=["frontend", "backend"])
    # La observasjonsøvelsen velge samme sampler i begge prosesser.
    parser.add_argument("--sampler", choices=["all", "ratio", "parent-ratio"], default="all")
    # Tillat en annen lokal port, blant annet i testene.
    parser.add_argument("--port", type=int)
    # La frontend vite hvilken backend-adresse den skal kalle.
    parser.add_argument("--backend-url", default="http://127.0.0.1:8092/lookup")
    # Les rolle og nettverksvalg.
    args = parser.parse_args()
    # Vis INFO-meldinger som inneholder sporingsbevis.
    logging.basicConfig(level=logging.INFO)
    # Klargjør eksport før serveren tar imot kall.
    provider, meter_provider = setup(args.role, args.sampler)
    # Opprett traceren som lager labens spans.
    tracer = trace.get_tracer("ai200.network")
    # Opprett instrumentene for labens målinger.
    meter = metrics.get_meter("ai200.network")
    # Tell forespørsler per status.
    requests = meter.create_counter("workshop.requests")
    # Samle svartid i millisekunder.
    duration = meter.create_histogram("workshop.duration", unit="ms")

    # Definer hvordan denne tjenesten håndterer HTTP-kall.
    class Handler(BaseHTTPRequestHandler):
        # Behandle én GET-forespørsel.
        def do_GET(self):
            # Start målingen av hele forespørselen.
            started = time.perf_counter()
            # Skill URL-sti fra spørringsparametere.
            parsed = urlsplit(self.path)
            # Les parameterne som styrer forsinkelse og kontrollert feil.
            query = parse_qs(parsed.query)
            # Anta et vellykket kall frem til en feil oppstår.
            status = 200
            # Start en serverspan for den innkommende forespørselen.
            with tracer.start_as_current_span(args.role, kind=SpanKind.SERVER,
                                             # Les sporingskontekst fra HTTP-headerne så backend kan fortsette samme trace.
                                             context=propagate.extract({key.lower(): value for key, value in self.headers.items()})) as span:
                # La kontrollerte nettverks- og backend-feil gi HTTP 500.
                try:
                    # Frontend videresender arbeidet til en annen prosess.
                    if args.role == "frontend":
                        # Legg en klientspan under frontendens serverspan.
                        with tracer.start_as_current_span("call-backend", kind=SpanKind.CLIENT):
                            # Lag headerne som skal følge det utgående kallet.
                            headers = {}
                            # Legg aktiv trace-kontekst i headerne før HTTP-kallet.
                            propagate.inject(headers)
                            # Send de samme delay-/fail-parameterne videre.
                            url = args.backend_url + "?" + parsed.query
                            # Gjør et ekte nettverkskall med sporingsheaderne.
                            with urllib.request.urlopen(urllib.request.Request(url, headers=headers), timeout=10) as response:
                                # Les svaret ferdig før klientspannen avsluttes.
                                response.read()
                    # Backend utfører det simulerte oppslaget.
                    else:
                        # Avgrens den valgte forsinkelsen til mellom null og ett sekund.
                        time.sleep(min(max(float(query.get("delay", ["0.02"])[0]), 0), 1))
                        # Bruk fail=1 til å slå på en kjent feil.
                        if query.get("fail", ["0"])[0] == "1":
                            # La feilen bli synlig både i backend og i klientkallet fra frontend.
                            raise RuntimeError("Controlled backend failure")
                # Fang feilen for å sende et kontrollert HTTP-svar.
                except Exception as error:
                    # Merk HTTP-svaret som en serverfeil.
                    status = 500
                    # Legg unntaket inn som en hendelse på spannen.
                    span.record_exception(error)
                    # Merk spannen som mislykket.
                    span.set_status(Status(StatusCode.ERROR, str(error)))
                # Ta med HTTP-statusen som span-attributt.
                span.set_attribute("http.response.status_code", status)
                # Gjør tjenestens rolle synlig på spannen.
                span.set_attribute("workshop.role", args.role)
                # Formater trace-ID slik at du kan sammenligne den mellom prosessene.
                trace_id = format(span.get_span_context().trace_id, "032x")
                # Send rolle, trace-ID og resultat til klienten.
                body = json.dumps({"role": args.role, "trace_id": trace_id, "status": status}).encode()
                # Start HTTP-svaret med resultatstatusen.
                self.send_response(status)
                # Oppgi at innholdet er JSON.
                self.send_header("Content-Type", "application/json")
                # Oppgi svarlengden i byte.
                self.send_header("Content-Length", str(len(body)))
                # Avslutt headerne før body sendes.
                self.end_headers()
                # Skriv svarinnholdet til klienten.
                self.wfile.write(body)
                # Vis samme trace-ID i begge tjenestenes logger.
                logging.info("role=%s trace_id=%s status=%s", args.role, trace_id, status)
            # Skill målingene etter HTTP-status.
            attributes = {"status": str(status)}
            # Tell denne forespørselen.
            requests.add(1, attributes)
            # Registrer hele forespørselens varighet i millisekunder.
            duration.record((time.perf_counter() - started) * 1000, attributes)

    # Bruk forskjellige standardporter for de to prosessene.
    port = args.port or (8091 if args.role == "frontend" else 8092)
    # Lytt bare lokalt på maskinen.
    server = ThreadingHTTPServer(("127.0.0.1", port), Handler)
    # La tjenesten kjøre frem til den stoppes.
    try:
        # Ta imot og behandle HTTP-kall.
        server.serve_forever()
    # Håndter Ctrl+C som en vanlig avslutning.
    except KeyboardInterrupt:
        # Fortsett til oppryddingen under.
        pass
    # Lukk serveren og tøm eksportkøene også ved avbrudd.
    finally:
        # Frigi lytteporten.
        server.server_close()
        # Forsøk å sende gjenværende spans før prosessen avsluttes.
        provider.force_flush()
        # Forsøk å sende gjenværende målinger.
        meter_provider.force_flush()
        # Stopp sporingsleverandøren.
        provider.shutdown()
        # Stopp måleleverandøren.
        meter_provider.shutdown()


# Start bare når filen kjøres direkte.
if __name__ == "__main__":
    # Kjør tjenesten med valgt rolle.
    main()

What does an OpenTelemetry TracerProvider own?

It creates tracers and holds their shared tracing setup: service metadata, the sampler and span processors. Tracers create spans; processors and exporters handle sending completed spans to a destination such as Azure Monitor.

m-64-1-1 · Åpne dette flashcardet

What sends OpenTelemetry spans to Azure Monitor?

An Azure Monitor trace exporter, such as Python AzureMonitorTraceExporter. It sends completed spans to the configured Application Insights destination. A span processor connects it to the tracer provider; a supported Azure Monitor distro can set up that pipeline for you.

m-64-2-1 · Åpne dette flashcardet

How do BatchSpanProcessor and SimpleSpanProcessor export spans?

SimpleSpanProcessor exports a completed sampled span synchronously, so export can delay the ending thread. BatchSpanProcessor queues sampled spans and normally exports them in background batches, reducing per-span overhead. Configure shutdown/flush for short-lived processes so queued telemetry has a chance to leave before the process exits.

m-64-3-1 · Åpne dette flashcardet

Which Python OpenTelemetry call starts an active span?

Use a context manager with a configured tracing provider: from opentelemetry import trace tracer = trace.get_tracer(__name__) with tracer.start_as_current_span("check-cache") as span: span.set_attribute("cache.hit", True) The with block makes this span current for nested operations and ends it when the block exits. Put the actual timed work inside that block. Starting a span without a configured recording/export pipeline does not make it appear in Azure Monitor.

m-64-4-1 · Åpne dette flashcardet

What must an application do to produce visible OpenTelemetry traces?

Configure a tracing provider and exporter, then create spans through library instrumentation or tracing code. Deploy the required packages and destination settings, restart the service, and run the code you want to trace. If traces are missing, check sampling, export failures and ingestion delay.

m-65-1-1 · Åpne dette flashcardet

How does the Azure Monitor Python exporter map span kinds to telemetry tables?

SERVER and CONSUMER spans become request telemetry: requests or AppRequests. CLIENT, PRODUCER and INTERNAL spans become dependency telemetry: dependencies or AppDependencies. The table names depend on query scope. INTERNAL usually represents work inside the process. Choose the span kind for the operation it describes.

m-65-5-1 · Åpne dette flashcardet

Which W3C header carries trace context between HTTP services?

traceparent, defined by the World Wide Web Consortium (W3C) Trace Context standard. It carries the trace ID, sending span ID and trace flags so the receiving service can continue the same distributed trace. Instrumentation libraries normally inject and read this header.

m-68-1-1 · Åpne dette flashcardet

Does creating a span automatically propagate context to another process?

No. The current span/context exists inside the process. A caller must inject its context into outgoing headers or message metadata, and the receiver must extract it before starting the child operation. Supported HTTP or messaging instrumentation often does this automatically. Without propagation, each service may create an unrelated trace.

m-68-2-1 · Åpne dette flashcardet

How is OpenTelemetry context propagated through an outbound Python HTTP call?

With tracing configured, enable the requests library's OpenTelemetry instrumentation before making calls: import requests from opentelemetry.instrumentation.requests import RequestsInstrumentor RequestsInstrumentor().instrument() requests.get("https://example.com/api", timeout=10) The instrumentation creates a client span and injects trace headers into the request. The receiving service must extract them, usually through its own instrumentation, to continue the same trace.

Which OpenTelemetry sampler honors a parent's sampling decision?

ParentBased. It uses the parent’s sampled/not-sampled state according to its configured local/remote parent rules. Its root sampler makes the decision when no parent exists. For example, ParentBased(root=TraceIdRatioBased(0.1)) starts about 10% of new root traces and normally follows parent decisions for their children.

m-66-5-1 · Åpne dette flashcardet

How does TraceIdRatioBased choose which traces to retain?

It makes a deterministic comparison based on the trace ID and the configured probability. With a ratio of 0.1, about 10% of suitable randomly generated trace IDs are retained over time. It does not simply keep every tenth request or guarantee exactly 10% of a small batch.

m-66-5-2 · Åpne dette flashcardet

Which Python OpenTelemetry sampler records all spans during local testing?

ALWAYS_ON, from opentelemetry.sdk.trace.sampling. Assign it as the TracerProvider's sampler to record and sample each span it evaluates. This is useful when debugging locally. You still need an exporter to send those spans to a viewer; it cannot restore spans dropped in another service.

m-67-1-1 · Åpne dette flashcardet

Which OpenTelemetry sampler keeps approximately 10% of traces?

TraceIdRatioBased(0.1) selects approximately 10% of trace IDs. For a distributed trace, ParentBased(root=TraceIdRatioBased(0.1)) is often used so root traces make the ratio decision and child spans follow their parent. This is probabilistic selection over traffic, not an exact count per minute.

m-67-2-1 · Åpne dette flashcardet

Where is an OpenTelemetry sampler configured?

On the TracerProvider. For example, after importing the SDK classes: TracerProvider(sampler=TraceIdRatioBased(0.1)) The provider asks the sampler whether to record/sample each new span. Processors handle completed spans, and exporters send them to a destination.

m-67-3-1 · Åpne dette flashcardet

Are metrics reduced by trace sampling?

No. Trace sampling chooses which spans or traces to retain. Metric measurements are collected and aggregated independently, so a request can count in a request-rate metric even when its trace is not retained.

What does configure_azure_monitor configure automatically?

For the Azure Monitor Python distro, configure_azure_monitor sets up supported trace, metric and logging providers, Azure exporters and supported library instrumentation. Supply the destination through the supported connection-string configuration and initialize it once. Avoid also manually instrumenting the same library, which can create duplicate spans. Available instrumentation depends on the installed distro version and packages.

m-65-2-1 · Åpne dette flashcardet

What are the Python names of common OpenTelemetry samplers?

Python provides ALWAYS_ON and ALWAYS_OFF instances, TraceIdRatioBased(rate), and ParentBased(root=...). For example, TraceIdRatioBased(0.1) selects roughly 10% of trace IDs. These Python identifiers differ from specification names and environment-variable values; use the spelling required by the API you configure.

m-66-2-1 · Åpne dette flashcardet

Which standard OTEL_TRACES_SAMPLER values select ratio sampling?

traceidratio and parentbased_traceidratio for OTEL_TRACES_SAMPLER. Set OTEL_TRACES_SAMPLER_ARG to a ratio such as 0.1 for about 10%. The parent-based form uses that ratio for root traces while respecting the configured parent rules. Vendor distros may support additional aliases, so distinguish standard names from vendor-specific options.

m-66-3-1 · Åpne dette flashcardet

Which Azure Monitor-specific sampler environment settings are available?

Supported Azure Monitor configurations use OTEL_TRACES_SAMPLER=microsoft.fixed_percentage with OTEL_TRACES_SAMPLER_ARG set to a fraction such as 0.1, or microsoft.rate_limited with the argument set to an approximate traces-per-second target such as 5.0. Set them before initializing telemetry. The two arguments have different units: 0.1 is a ratio, while 5.0 is a rate.

m-66-4-1 · Åpne dette flashcardet

What rate-limited sampling default is used by recent Azure Monitor Python distros?

For azure-monitor-opentelemetry starting at version 1.8.6, configure_azure_monitor defaults to rate-limited sampling targeting approximately five traces per second when no sampler options override it. This is a version-specific default and an approximate local rate, not five spans total across a distributed application. Check the installed distro and any environment-variable overrides.

m-66-6-1 · Åpne dette flashcardet

How does Azure Monitor fixed-percentage sampling help coordinate trace selection across services?

It bases selection on the trace ID. Services using compatible logic, the same percentage and the same propagated trace context can therefore keep or drop the same trace. In the Azure Monitor Python distro, sampling_ratio=0.1 requests about 10%. Mixing incompatible sampling strategies can leave incomplete distributed traces.

Finn feilrate, treghet og relaterte kall

  1. Åpne queries.kql og kjør én spørring om gangen i Logs. Start med datatable-øvelsen, som ikke avhenger av at telemetri har kommet inn.
  2. Forutsi resultatene av where, extend, project, distinct, summarize, join og render før du kjører dem. Sammenlign inner og standard innerunique når venstre side har flere rader med samme nøkkel.
  3. Start begge tjenestene med Azure-tilkobling, generer nye kall og bruk spørringen for AppRequests/AppDependencies i workspace-kontekst. Ved Application Insights-kontekst bruker du den alternative requests/dependencies-varianten. Oppgi varighetsenhet og bruk riktig rolle-/operation-felt.
  4. Lag tidsbøtter med bin, avgrens med ago og finn ett feilende end-to-end-kall. Skriv hvorfor tidsfilter og riktig kildetabell hører tidlig i spørringen.

// SPØRRING 1: Syntetiske data gjør fasiten uavhengig av ventetid på Azure-telemetri.
let observations = datatable(OperationId:string, Role:string, DurationMs:real, Success:bool, AgeMinutes:int)
// Fire forespørsler, hvor to feiler.
[
// Første rad er et vellykket kall.
"a", "frontend", 20.0, true, 1,
// Andre rad er et mislykket kall med samme operasjons-ID.
"a", "frontend", 30.0, false, 2,
// Tredje rad viser en treg og feilende backend.
"b", "backend", 2500.0, false, 3,
// Fjerde rad er et nytt vellykket kall.
"c", "frontend", 10.0, true, 4
];
// Bruk datasettet som kildetabell.
observations
// Lag en aktuell tidskolonne fra den relative alderen.
| extend TimeGenerated = now() - AgeMinutes * 1m
// Avgrens tid før aggregeringen.
| where TimeGenerated > ago(30m)
// Velg de kolonnene resten av spørringen trenger.
| project OperationId, Role, DurationMs, Success
// Tell alle og feilende rader og beregn en percentil.
| summarize Requests = count(), Failures = countif(Success == false), P95Ms = percentile(DurationMs, 95)
// To av fire gir 50 prosent i denne syntetiske testen.
| extend FailureRate = 100.0 * Failures / Requests

// SPØRRING 2: En standard inner-join beholder begge a-radene på venstre side.
let incoming = datatable(OperationId:string, Attempt:int)["a",1,"a",2,"b",1];
// Høyre side har én komponent per operasjon.
let components = datatable(OperationId:string, Component:string)["a","frontend","b","backend"];
// Start på venstre side med tre rader.
incoming
// Fjern kind=inner og sammenlign med standard innerunique: to rader i stedet for tre.
| join kind=inner components on OperationId
// Vis bare nøkkel, forsøk og komponent.
| project OperationId, Attempt, Component

// SPØRRING 3: Grupper tidsstempler og be om en visualisering.
let observations = datatable(AgeMinutes:int, Role:string)[1,"frontend",2,"frontend",3,"backend"];
// Start med de tre relative tidspunktene.
observations
// Lag faktiske tidsstempler ved kjøring.
| extend TimeGenerated = now() - AgeMinutes * 1m
// Tell rader per femminuttersbøtte og rolle.
| summarize Requests = count() by bin(TimeGenerated, 5m), Role
// Be klienten vise resultatet som tidsdiagram.
| render timechart

// SPØRRING 4: Skill unike verdier fra alle inngående rader.
datatable(Role:string)["frontend","frontend","backend"]
// Resultatet skal ha to roller.
| distinct Role

// SPØRRING 5: Kjør i Log Analytics-workspace etter at egen telemetri er ingestert.
AppRequests
// Bruk riktig tidskolonne for workspace-skjemaet.
| where TimeGenerated > ago(30m)
// Avgrens til de to tjenestenavnene fra services.py.
| where AppRoleName startswith "ai200-"
// Mål per komponent, med varighet i millisekunder.
| summarize Requests=count(), Failures=countif(Success == false), P95Ms=percentile(DurationMs,95) by AppRoleName

// SPØRRING 6: Alternativet brukes i Application Insights sin spørringskontekst.
requests
// Samme tidsfilter har et annet kolonnenavn her.
| where timestamp > ago(30m)
// Filtrer på rollen i Application Insights-skjemaet.
| where cloud_RoleName startswith "ai200-"
// Samme signal har andre kolonnenavn, men samme varighetsenhet.
| summarize Requests=count(), Failures=countif(success == false), P95Ms=percentile(duration,95) by cloud_RoleName

// SPØRRING 7: Finn relaterte requests og dependencies i workspace-skjemaet.
AppRequests
// Avgrens venstre tabell tidlig.
| where TimeGenerated > ago(30m) and AppRoleName startswith "ai200-"
// Koble på dependencies med samme OperationId.
| join kind=leftouter (
    // Velg den tilsvarende workspace-tabellen.
    AppDependencies
    // Avgrens også høyre tabell før join.
    | where TimeGenerated > ago(30m)
    // Gi avhengighetsfeltene tydelige resultatnavn.
    | project OperationId, DependencyName=Name, DependencyMs=DurationMs
// Avslutt høyre uttrykk og angi nøkkelen.
) on OperationId
// Velg bare felter som hjelper å forklare det distribuerte kallet.
| project TimeGenerated, OperationId, AppRoleName, Success, DurationMs, DependencyName, DependencyMs

What does KQL where do to its input rows?

It keeps rows whose condition is true. For example: AppRequests | where Success == false This returns failed request records and leaves their columns available. The | pipe passes the table result into the next operation. KQL means Kusto Query Language; filter is an equivalent operator name.

What rows does KQL summarize return?

It combines input rows into one summary row per group in the by clause. For example: AppRequests | summarize Requests=count() by AppRoleName This counts requests for each service name. Without a by clause, it produces one summary row for the entire input.

m-60-2-1 · Åpne dette flashcardet

What does KQL project do to columns?

It chooses the output columns and can rename or calculate them; unlisted columns are dropped. For example: AppRequests | project Service=AppRoleName, DurationSeconds=DurationMs / 1000.0 The result has Service and DurationSeconds only. Use extend instead when you want to retain other columns while calculating a value.

m-60-3-1 · Åpne dette flashcardet

What does KQL extend do to columns?

It adds a calculated column, or replaces an existing column with that name, while retaining the other columns and individual rows. For example: AppRequests | extend DurationSeconds=DurationMs / 1000.0 This calculates seconds for each request. It does not group rows; use summarize for grouped aggregates such as an average per service.

m-60-4-1 · Åpne dette flashcardet

What does KQL distinct return?

It returns unique combinations of the selected column values. For example: AppRequests | distinct AppRoleName, ResultCode Each service/result-code pair appears once, even if thousands of requests share it. It does not calculate a count; use summarize count() by those columns when counts are also needed.

m-60-5-1 · Åpne dette flashcardet

What does KQL join do?

It combines rows from two tables when their key values match. For example: AppRequests | join kind=inner (AppDependencies) on OperationId This matches requests with dependency records from the same operation. kind=inner keeps matched rows only. Several matches for one key produce several output rows, so choose the key and join kind carefully.

m-60-5-2 · Åpne dette flashcardet

What does KQL render do?

It asks the query client to visualize the tabular result. For example: AppRequests | summarize Requests=count() by bin(TimeGenerated, 5m) | render timechart The query first produces a count per five-minute time bucket; render then requests a time chart. Rendering does not itself group data or change the underlying stored records.

m-60-5-3 · Åpne dette flashcardet

In what order should time filters, joins and aggregation usually run in a KQL query?

Start with the relevant tables and apply selective time and other filters early, including on both sides of a join when appropriate. Join only the needed rows/columns, then aggregate at the level required by the answer. Sometimes pre-aggregating before a join is better; the key principle is reducing processed data without changing the intended result.

Which table and column names are used in the Application Insights query schema?

In Application Insights resource queries, common tables are requests, dependencies, traces and exceptions. Fields include timestamp, duration, success, operation_Id, operation_Name and cloud_RoleName. Names are case-sensitive. Despite its name, traces stores application log records; distributed-trace spans are normally in requests and dependencies.

m-61-1-1 · Åpne dette flashcardet

Which table and column names are used in the workspace-based query schema?

In workspace queries, use AppRequests, AppDependencies, AppTraces and AppExceptions. Common fields include TimeGenerated, DurationMs, Success, OperationId, OperationName and AppRoleName. These correspond to the Application Insights resource-query names. Choose the schema for the scope you opened instead of mixing timestamp with AppRequests or TimeGenerated with requests.

What unit is used for request duration in Application Insights queries?

Milliseconds. In the Application Insights query schema, requests.duration > 2000 means longer than two seconds: requests | where duration > 2000 The workspace-schema equivalent is: AppRequests | where DurationMs > 2000

m-61-3-1 · Åpne dette flashcardet

Which telemetry field identifies the emitting service?

cloud_RoleName in the Application Insights query schema, or AppRoleName in workspace tables. It identifies the logical emitting service, such as orders-api. The role-instance field identifies a particular running instance. Give services distinct, stable role names so queries and Application Map can separate them.

m-61-4-1 · Åpne dette flashcardet

How does KQL express the last 30 minutes?

Use ago(30m), meaning the current UTC time minus thirty minutes. A workspace query is: AppRequests | where TimeGenerated > ago(30m) In Application Insights resource queries, use timestamp instead. The m is a timespan unit, not a variable; other examples include 10s, 1h and 2d.

m-62-1-1 · Åpne dette flashcardet

How does KQL bin assign timestamps to time buckets?

bin rounds each timestamp down to the start of a fixed interval. For example: AppRequests | summarize Requests=count() by bin(TimeGenerated, 5m) | render timechart Requests from 10:00 through just before 10:05 share the 10:00 bucket. The chart uses the datetime bucket as its time axis. Empty intervals need explicit padding if they should appear as zero.

m-62-3-1 · Åpne dette flashcardet

Which KQL join kind is used when no kind is specified?

innerunique. It deduplicates the left input by the join key before matching, so repeated left-side keys do not behave like an ordinary inner join. Specify kind=inner when you need every matching pair. For performance, often put the smaller suitable input on the left, but do not swap inputs without considering how that changes the join’s meaning.

Log Analytics in the Azure portal. How do Azure Monitor metrics differ from logs?

Metrics are numeric time series, such as CPU percentage or request counts over time, suitable for trend charts and threshold alerts. Logs are timestamped records with fields that can be searched, joined and aggregated using KQL. An application can emit both: a metric for the failure rate and log records explaining individual failures.

What type of operations does the Azure Activity Log record?

Management operations on Azure resources, such as creating a virtual machine, changing a vault's configuration or deploying infrastructure. These are control-plane events. Use application telemetry to investigate what the running app did with a user request.

m-63-2-1 · Åpne dette flashcardet

Why start a KQL query with the relevant table?

It selects the input data for the query. For example, AppRequests starts with request records, while AzureActivity starts with resource-management events. Choosing the relevant table avoids searching unrelated logs and makes later filters/aggregates meaningful. Time and other filters should normally narrow that input early.

m-63-4-1 · Åpne dette flashcardet

How do KQL == and =~ differ?

For string equality, == is case-sensitive and =~ is case-insensitive. For example, "Prod" == "prod" is false, while "Prod" =~ "prod" is true. Prefer == when the intended comparison is case-sensitive; using =~ does not make table or column identifiers case-insensitive.

m-63-4-2 · Åpne dette flashcardet

What routes supported Azure resource platform logs to Log Analytics, Storage or Event Hubs?

A diagnostic setting on the Azure resource. Select the supported log categories/category groups and destinations, such as a Log Analytics workspace, Storage account or Event Hub. It routes the resource’s platform logs; it does not automatically add application-level tracing or collect every possible category without configuration.

Which Log Analytics table contains Azure Activity Log records when collected?

AzureActivity, after the Activity Log has been routed to that Log Analytics workspace. For example: AzureActivity | where TimeGenerated > ago(1h) The Activity Log exists at the Azure subscription level by default, but its records are not automatically present in every workspace you open.

Why can a telemetry ingestion daily cap hide later incidents?

When the cap is reached, eligible log ingestion can stop until the collection period resets, creating gaps that hide later failures. Some data may be exempt and enforcement can overshoot, so the cap is not a precise spending limit. Use volume alerts and collection/sampling controls as well, rather than relying on a cap as the normal cost-management method.

Which KQL function references another Log Analytics workspace?

workspace(...). For example: workspace("workspace-resource-id").AppRequests | where TimeGenerated > ago(1h) Replace the string with a supported identifier for the other workspace. You need access to that workspace and must follow cross-resource query limits. The function selects remote log data; it does not copy it into the current workspace.

Which Application Insights query table stores custom metric telemetry?

customMetrics in the Application Insights resource-query schema, or AppMetrics in workspace queries. This is standalone custom metric telemetry, such as an application-defined queue-length measurement. Numeric measurements attached to individual requests/events can instead live in those records’ measurement fields, so not every custom numeric property belongs in customMetrics.
