"""Local workshop API. In-memory deduplication is deliberately not production storage."""

# Serialiser svarene som JSON.
import json
# Skriv hendelser og kontrollerte feil til Functions-loggen.
import logging
# Les appinnstillingene fra miljøet.
import os
# Beskytt det delte minnesettet mot samtidige forespørsler.
import threading

# Bruk Functions-dekoratorer og HTTP-typer.
import azure.functions as func

# Krev en Function-nøkkel på vanlige HTTP-ruter i Azure.
app = func.FunctionApp(http_auth_level=func.AuthLevel.FUNCTION)
# Husk behandlede dokument-ID-er bare så lenge denne prosessen kjører.
processed = set()
# La kontroll og registrering av en ID skje samlet.
processed_lock = threading.Lock()


# Ta imot dokumenter på /api/documents.
@app.route(route="documents", methods=["POST"])
# Valider dokumentet og demonstrer håndtering av samme ID flere ganger.
def documents(req: func.HttpRequest) -> func.HttpResponse:
    # Forsøk å lese forespørselsinnholdet som JSON.
    try:
        # Konverter JSON-body til Python-data.
        payload = req.get_json()
    # Håndter innhold som ikke kan tolkes som JSON.
    except ValueError:
        # Svar med klientfeil for ugyldig JSON.
        return func.HttpResponse("Body must be JSON", status_code=400)
    # Krev et JSON-objekt, ikke for eksempel en liste.
    if not isinstance(payload, dict):
        # Forklar hvorfor datatypen ble avvist.
        return func.HttpResponse("Body must be a JSON object", status_code=400)
    # Hent dokument-ID-en som brukes til duplikatkontroll.
    document_id = payload.get("id")
    # Avvis manglende, tomme og ikke-tekstlige ID-er.
    if not isinstance(document_id, str) or not document_id.strip():
        # Send en tydelig valideringsfeil til klienten.
        return func.HttpResponse("id must be a non-empty string", status_code=400)
    # Unngå at to samtidige kall registrerer samme ID som ny.
    with processed_lock:
        # Kontroller om denne prosessen allerede har sett ID-en.
        duplicate = document_id in processed
        # Registrer ID-en før en annen tråd får tilgang.
        processed.add(document_id)
    # Gjør duplikathåndteringen synlig i loggen.
    logging.info("document_received duplicate=%s", duplicate)
    # Returner et JSON-svar til klienten.
    return func.HttpResponse(
        # Vis om dette kallet ble behandlet som nytt.
        json.dumps({"id": document_id, "processed": not duplicate}),
        # Angi riktig innholdstype for JSON-responsen.
        mimetype="application/json",
    )


# La Key Vault-øvelsen kontrollere om hemmeligheten ble levert til appen.
@app.route(route="configuration", methods=["GET"])
# Rapporter bare status for konfigurasjonen.
def configuration(req: func.HttpRequest) -> func.HttpResponse:
    # Les hemmeligheten som appinnstilling.
    secret = os.getenv("DEMO_SECRET", "")
    # Skill en faktisk verdi fra en Key Vault-referanse som ikke er løst.
    resolved = bool(secret) and not secret.startswith("@Microsoft.KeyVault(")
    # Returner et JSON-svar til klienten.
    return func.HttpResponse(
        # Send bare en boolsk status, aldri hemmelighetens verdi.
        json.dumps({"secret_configured": resolved}),
        # Angi riktig innholdstype for JSON-responsen.
        mimetype="application/json",
    )


# Gi labens Event Grid-webhook en HTTP-rute uten Function-nøkkel.
@app.route(route="event-hook", methods=["POST"], auth_level=func.AuthLevel.ANONYMOUS)
# Håndter valideringshendelser og vanlige Event Grid-hendelser.
def event_hook(req: func.HttpRequest) -> func.HttpResponse:
    # Les hendelseslisten i Event Grid Schema-format.
    events = req.get_json()
    # Krev en liste slik dette hendelsesskjemaet bruker.
    if not isinstance(events, list):
        # Avvis et annet meldingsformat.
        return func.HttpResponse("Expected Event Grid Schema array", status_code=400)
    # Gå gjennom hendelsene i leveransen.
    for event in events:
        # Gjenkjenn håndtrykket som validerer webhook-abonnementet.
        if event.get("eventType") == "Microsoft.EventGrid.SubscriptionValidationEvent":
            # Send valideringskoden tilbake i validationResponse.
            return func.HttpResponse(json.dumps({"validationResponse": event["data"]["validationCode"]}),
                                     # Svar med JSON slik valideringen forventer.
                                     mimetype="application/json")
    # Bruk en appinnstilling til å slå på en kontrollert leveringsfeil.
    if os.getenv("FAIL_EVENTS", "false").lower() == "true":
        # Gjør den fremprovoserte feilen synlig i loggen.
        logging.warning("Controlled Event Grid delivery failure")
        # Svar med 503 så Event Grid kan forsøke levering igjen.
        return func.HttpResponse("Retry this delivery", status_code=503)
    # Gå gjennom hendelsene i leveransen.
    for event in events:
        # Vis hvilke hendelser som kom gjennom abonnementets filter.
        logging.info("event id=%s subject=%s", event.get("id"), event.get("subject"))
    # Bekreft vellykket mottak med HTTP 200.
    return func.HttpResponse("Accepted")


# Ta imot HTTP uten å vente på at bakgrunnsarbeidet blir ferdig.
@app.route(route="enqueue", methods=["POST"])
# Send bodyen til en Storage-kø gjennom en output-binding.
@app.queue_output(arg_name="message", queue_name="documents", connection="AzureWebJobsStorage")
# Denne funksjonen har bare HTTP-triggeren; køtriggeren er separat.
def enqueue(req: func.HttpRequest, message: func.Out[str]) -> func.HttpResponse:
    # Overlat selve sendingen til Functions-hosten.
    message.set(req.get_body().decode("utf-8"))
    # Aksept betyr at arbeidet er lagt til flyten, ikke at det er ferdig behandlet.
    return func.HttpResponse("Queued", status_code=202)


# Behandle meldinger fra samme Storage-kø som output-bindingen bruker.
@app.queue_trigger(arg_name="message", queue_name="documents", connection="AzureWebJobsStorage")
# La en kontrollert feil gjøre hostens retry- og poison-atferd synlig.
def background(message: func.QueueMessage):
    # Les bare den ufarlige testbodyen.
    body = message.get_body().decode("utf-8")
    # Marker akkurat den valgte feilhendelsen.
    if "fail" in body.lower():
        # Hosten håndterer nye forsøk og eventuell poison queue.
        raise ValueError("Controlled Storage queue processing failure")
    # Bekreft behandling uten å logge eksterne credentials.
    logging.info("Background work completed: %s", message.id)


# Timer-formatet har seks felter og kjører hvert femte minutt.
@app.timer_trigger(schedule="0 */5 * * * *", arg_name="timer", run_on_startup=False, use_monitor=True)
# Vis en koordinert planlagt kjøring i hostloggen.
def scheduled(timer: func.TimerRequest):
    # Hosten bruker Storage til å holde rede på tidsplanen.
    logging.info("Scheduled workshop task, past_due=%s", timer.past_due)

What is local.settings.json used for?

local.settings.json supplies settings used when running Functions locally, including environment-variable values. Azure deployments use app settings instead. Because local settings can contain secrets, verify that the repository ignores this file before adding real values.

What is host.json used for in Azure Functions?

host.json configures the Functions host across a Function App, including logging, timeouts and extension-specific trigger behavior. It is deployed with the code. Keep environment-specific secrets and connection values in the appropriate app settings.

How do Anonymous, Function and Admin HTTP authorization levels differ?

Anonymous requires no Functions key. Function accepts a function-specific key or a host key. Admin requires the host’s master key. A required key can be supplied through x-functions-key or the code query parameter. These shared keys control access but do not establish an individual user’s identity.

Eksempel · HTTP request header

x-functions-key: <function-key>

For an endpoint requiring Function authorization, replace the placeholder with an authorized function or host key. This shared endpoint credential does not identify a particular end user.

Are Function keys enforced by the local Functions host?

Normally not when running directly through local Core Tools. The deployed Azure endpoint enforces its configured authorization level; local container hosting is a documented exception that still requires keys. A keyless local test therefore does not establish that the deployed endpoint allows anonymous callers.

Eksempel · Scenario

An endpoint configured with Function authorization accepts a keyless request when tested directly through local Core Tools. Its deployed Azure endpoint still requires the key; that local success did not prove anonymous production access.

How does an HTTP trigger's methods list restrict accepted requests?

It selects which HTTP methods may invoke that function. For example, methods=["POST"] permits POST for the route while excluding GET. If omitted, all supported methods are accepted. Authorization is configured separately.

What does an Azure Functions output binding do?

An output binding sends or writes a handler’s output to a configured service. For example, it can write a queue message supplied by the function. The binding needs the destination configuration and permission to write there.

m-49-3-1 · Åpne dette flashcardet

What happens after Queue Storage-triggered processing repeatedly fails?

The default maximum is five attempts in total, including the first. If they all fail, the Functions runtime adds the message to <original-queue-name>-poison. maxDequeueCount in host.json controls the limit. Monitor and handle the poison queue so failed work has a recovery path.

Eksempel · JSON · host.json

{"version":"2.0","extensions":{"queues":{"maxDequeueCount":3}}}

This example changes a Storage queue trigger’s attempt limit to three, including the first attempt. If all fail, the host uses the original queue’s -poison queue; three is not the default.

What schedule format does an Azure Functions Timer trigger use?

Use an NCRONTAB schedule, normally with six fields: second, minute, hour, day, month and day of week. For example, 0 */5 * * * * runs at second zero every five minutes. UTC is the default; time-zone customization depends on the hosting configuration.

Which Functions trigger runs on a fixed schedule?

How should slow HTTP-request work be moved into background processing?

Save a job or enqueue a message, then return HTTP 202 Accepted with its identifier or status URL. A background worker performs the slow operation and records the result. The caller checks status separately rather than holding the original request open.

Eksempel · HTTP response example

HTTP/1.1 202 Accepted
Location: /jobs/42
Content-Type: application/json

{"jobId":"42","status":"queued"}

Save or enqueue the work before returning this response. The client checks GET /jobs/42 later while a background worker completes the operation.

Can an Azure Function have multiple optional bindings?

Yes. A function has one trigger and can also have several input and output bindings. For example, an HTTP-triggered function can read a blob through an input binding and write a queue message through an output binding.

How do polling and Event Grid-based Blob triggers differ?

The polling trigger checks Storage for changes, so discovery can be delayed. The Event Grid-based Blob trigger receives event notifications and normally detects new work sooner. Design either handler to tolerate repeated delivery. Receiving a notification and successfully completing the business operation are separate steps.

What does an Azure Functions Cosmos DB trigger use to track progress?

It uses a lease container to store ownership and progress for change-feed ranges. The Functions host uses that state to distribute work and resume after restarts or ownership changes.

What limits the synchronous response time of an HTTP-triggered function?

The documented platform maximum for an HTTP-triggered response is 230 seconds, even if functionTimeout is longer. A shorter function, proxy or client timeout can stop the request sooner. For longer work, return promptly with a job identifier or status URL and continue processing in the background.

Why must identity permissions be configured before Key Vault references can resolve?

A reference identifies which secret to read; it does not grant access. The configured identity must have secret-read permission, and the vault must be reachable. Correct reference syntax alone cannot overcome missing authorization or blocked networking.

How does Event Grid space out push-delivery retries?

For classic push delivery, retries start after short delays and back off as failures continue. The documented schedule starts at 10 seconds, then 30 seconds and one minute, eventually reaching 12-hour intervals. The timing is best effort and includes randomization. The event lifetime and maximum-attempt policy can stop delivery sooner.

How long does Event Grid wait for a push endpoint response?

Event Grid waits 30 seconds. Without a response, it queues the event for another delivery attempt. A late success response might arrive after a retry is already queued, so the handler must tolerate duplicates. For slow work, safely enqueue the task and acknowledge promptly, then process it in the background.

m-38-3-1 · Åpne dette flashcardet

Which HTTP success codes acknowledge Event Grid push delivery?

HTTP 200, 201, 202, 203 and 204 count as successful delivery. Return one of these only after the handler has safely accepted the event, because Event Grid then considers its delivery complete. Other responses follow the retry or dead-letter rules; a 202 response does not make Event Grid track your later background processing.
