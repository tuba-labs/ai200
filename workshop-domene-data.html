-- Aktiver pgvector-utvidelsen i den valgte databasen.
CREATE EXTENSION IF NOT EXISTS vector;

-- Opprett labtabellen hvis den ikke finnes fra før.
CREATE TABLE IF NOT EXISTS workshop_notes (
    -- Gi hver rad en entydig identifikator.
    id integer PRIMARY KEY,
    -- Bruk topic som metadatafilter i søkene.
    topic text NOT NULL,
    -- Lagre teksten som et senere RAG-steg kan hente frem.
    content text NOT NULL,
    -- Lagre nøyaktig tre komponenter per undervisningsvektor.
    embedding vector(3) NOT NULL
);

-- Hand-written teaching vectors, not output from an embedding model.
-- Sett inn fire eksempler. Vektorene er skrevet for hånd, ikke generert av en modell.
INSERT INTO workshop_notes (id, topic, content, embedding) VALUES
    -- Første containerdokument peker langs første vektordimensjon.
    (1, 'containers', 'A container packages an application and its dependencies.', '[1,0,0]'),
    -- Andre containerdokument ligger nær det første i vektorrommet.
    (2, 'containers', 'Read container logs when an application fails to start.', '[0.9,0.1,0]'),
    -- Datadokumentet ligger nær spørsmålsvektoren, men kan filtreres bort av topic.
    (3, 'data', 'A database stores and retrieves application data.', '[0,1,0]'),
    -- Sikkerhetsdokumentet gir oss et eksempel i et annet emne.
    (4, 'security', 'Managed identity removes the need to store a credential.', '[0,0,1]')
-- Oppdater eksisterende rader når skriptet kjøres på nytt.
ON CONFLICT (id) DO UPDATE SET
    -- Bruk emnet fra raden vi forsøkte å sette inn.
    topic = EXCLUDED.topic,
    -- Oppdater teksten til verdien i eksempeldatasettet.
    content = EXCLUDED.content,
    -- Oppdater også vektoren, og avslutt SQL-setningen.
    embedding = EXCLUDED.embedding;

postgres_client.py — vis fil og forklaring

Kildefil: labs/data/postgres_client.py

"""Run the same parameterized vector query with a direct connection and a pool."""
# Velg setup, query eller compare fra kommandolinjen.
import argparse
# Les tilkoblingen fra miljøet fremfor å lagre passord i filen.
import os
# Mål tiden brukt på direkte forbindelser og forbindelsespool.
import time
# Finn SQL-filen ved siden av dette skriptet.
from pathlib import Path

# Bruk PostgreSQL-driveren til å kjøre SQL.
import psycopg
# Registrer hvordan pgvector-verdier konverteres i forbindelsen.
from pgvector.psycopg import register_vector
# Gjenbruk et lite antall databaseforbindelser.
from psycopg_pool import ConnectionPool


# Kjør dette oppsettet på hver nye forbindelse i poolen.
def configure(connection):
    # Lær forbindelsen å håndtere databasens vector-type.
    register_vector(connection)
    # Avslutt transaksjonen fra typeoppsettet før forbindelsen brukes av poolen.
    connection.commit()


# Bruk samme spørring i begge målingene.
def query(connection):
    # Kjør parameterisert SQL og hent de to nærmeste radene.
    return connection.execute(
        # Hent dokumentets ID, metadata og tekst.
        "SELECT id, topic, content FROM workshop_notes "
        # Filtrer på emne og sorter etter cosinusavstand med pgvector.
        "WHERE topic = %s ORDER BY embedding <=> %s::vector LIMIT 2",
        # Send verdiene separat fra SQL-teksten.
        ("containers", "[0,1,0]"),
    ).fetchall()


# Samle oppsett, spørring og sammenligning i ett kjørbart eksempel.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Tillat bare de tre laboperasjonene.
    parser.add_argument("mode", choices=["setup", "query", "compare"])
    # Les operasjonen deltakeren valgte.
    args = parser.parse_args()
    # Bruk tilkoblingsstrengen satt i terminalen.
    connection_string = os.environ["PGCONNINFO"]
    # Disable prepared statements so the example also works through transaction pooling.
    # Deaktiver automatisk server-side prepare for å støtte transaction pooling.
    options = {"prepare_threshold": None}
    # Opprett skjema og eksempeldata.
    if args.mode == "setup":
        # Åpne og lukk en direkte forbindelse for denne operasjonen.
        with psycopg.connect(connection_string, **options) as connection:
            # Les setup.sql fra samme mappe og kjør innholdet på serveren.
            connection.execute(Path(__file__).with_name("setup.sql").read_text(encoding="utf-8"))
        # Bekreft at oppsettet er fullført.
        print("Schema and four example rows ready")
    # Kjør ett vanlig oppslag.
    elif args.mode == "query":
        # Åpne og lukk en direkte forbindelse for denne operasjonen.
        with psycopg.connect(connection_string, **options) as connection:
            # Behandle radene fra den parameteriserte spørringen.
            for row in query(connection):
                # Vis resultatet i terminalen.
                print(row)
    # Sammenlign tidsbruk med og uten forbindelsespool.
    else:
        # Bruk samme antall operasjoner i begge målingene.
        count = 10
        # Start tidsmålingen for direkte forbindelser.
        started = time.perf_counter()
        # Åpne en ny forbindelse for hver repetisjon.
        for _ in range(count):
            # Inkluder forbindelsesopprettelsen i målingen.
            with psycopg.connect(connection_string, **options) as connection:
                # Kjør samme arbeid som i pool-varianten.
                query(connection)
        # Vis samlet tid for nye forbindelser.
        print(f"{count} new connections: {time.perf_counter() - started:.3f}s")
        # Opprett en pool med mellom én og tre forbindelser.
        with ConnectionPool(connection_string, min_size=1, max_size=3,
                            # Bruk samme tilkoblingsvalg og registrer pgvector for hver forbindelse.
                            kwargs=options, configure=configure) as pool:
            # Vent til poolen er klar før målingen starter.
            pool.wait()
            # Start målingen etter klargjøringen av poolen.
            started = time.perf_counter()
            # Gjenta samme antall spørringer.
            for _ in range(count):
                # Lån en forbindelse og lever den tilbake etter bruk.
                with pool.connection() as connection:
                    # Kjør den samme spørringen på en gjenbrukt forbindelse.
                    query(connection)
            # Vis tiden for spørringene gjennom poolen.
            print(f"{count} pooled queries: {time.perf_counter() - started:.3f}s")


# Start bare når filen kjøres direkte.
if __name__ == "__main__":
    # Kjør den valgte laboperasjonen.
    main()

"""Small isolated Cosmos DB lab; credentials are read from the environment."""
# Velg hvilken Cosmos-operasjon eksemplet skal utføre.
import argparse
# Les endpoint, nøkkel og databasenavn fra terminalens miljø.
import os

# Bruk Cosmos SDK-et og en eksplisitt partisjonsnøkkel.
from azure.cosmos import CosmosClient, PartitionKey

# Bruk små undervisningsdokumenter med håndskrevne vektorer.
DOCUMENTS = [
    # Første dokument ligger i partisjonen containers.
    {"id": "note-1", "topic": "containers", "text": "Package an application", "embedding": [1.0, 0.0, 0.0]},
    # Andre dokument ligger i samme partisjon, med en lignende vektor.
    {"id": "note-2", "topic": "containers", "text": "Inspect container logs", "embedding": [0.9, 0.1, 0.0]},
    # Tredje dokument viser hvordan metadatafilter kan utelate et nært vektortreff.
    {"id": "note-3", "topic": "data", "text": "Retrieve stored data", "embedding": [0.0, 1.0, 0.0]},
]


# La SDK-et sende responsheaderne til denne callbacken.
def report(headers, _body):
    # Hent kostnaden i request units fra responsheaderen.
    charge = headers.get("x-ms-request-charge")
    # Skriv bare RU når headeren finnes.
    if charge:
        # Vis grunnlaget for sammenligning av oppslag.
        print("Request charge:", charge, "RU")


# Kjør operasjonen som ble valgt på kommandolinjen.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Begrens valgene til operasjonene workshopene bruker.
    parser.add_argument("mode", choices=["setup", "query", "point", "vector-setup", "vector", "update"])
    # Les valgt operasjon.
    args = parser.parse_args()
    # Åpne klienten med labkontoens endpoint og nøkkel, og lukk den etter bruk.
    with CosmosClient(os.environ["COSMOS_ENDPOINT"], credential=os.environ["COSMOS_KEY"]) as client:
        # Use a database without shared throughput so vector search is supported.
        # Velg databasen; dette oppretter ikke databasen.
        database = client.get_database_client(os.getenv("COSMOS_DATABASE", "workshop"))
        # Hold vektorpolicyen i en egen container.
        name = "vectors" if args.mode.startswith("vector") else "notes"
        # Hent en klient for den valgte containeren.
        container = database.get_container_client(name)
        # Opprett labcontainer og eksempeldata ved setup.
        if args.mode in {"setup", "vector-setup"}:
            # Beskriv den vanlige indekseringspolicyen.
            policy = {
                # Oppdater indeksene automatisk når dokumentene endres.
                "indexingMode": "consistent", "automatic": True,
                # Indekser dokumentfeltene som standard.
                "includedPaths": [{"path": "/*"}],
                # Unnta vektorarrayet fra vanlig feltindeksering.
                "excludedPaths": [{"path": "/embedding/*"}],
            }
            # Legg bare til vektorinnstillinger for vektorcontaineren.
            extra = {}
            # Konfigurer en egen indeks for VectorDistance-søk.
            if args.mode == "vector-setup":
                # Bruk en flat vektorindeks på embedding-feltet.
                policy["vectorIndexes"] = [{"path": "/embedding", "type": "flat"}]
                # Beskriv hvilke vektorer embedding-feltet inneholder.
                extra["vector_embedding_policy"] = {"vectorEmbeddings": [{
                    # Lagre vektorene i embedding og behandle verdiene som float32.
                    "path": "/embedding", "dataType": "float32",
                    # Krev tre dimensjoner og sammenlign med cosinusavstand.
                    "dimensions": 3, "distanceFunction": "cosine",
                }]}
            # Opprett containeren hvis den mangler; eksisterende policy erstattes ikke her.
            container = database.create_container_if_not_exists(
                # Bruk topic som partisjonsnøkkel.
                id=name, partition_key=PartitionKey(path="/topic"),
                # Bruk policyen over og 400 RU/s dedikert throughput.
                indexing_policy=policy, offer_throughput=400, **extra,
            )
            # Skriv inn undervisningsdokumentene.
            for document in DOCUMENTS:
                # Opprett eller erstatt dokumentet, og vis operasjonens RU.
                container.upsert_item(document, response_hook=report)
            # Vis hvilken container som er klargjort.
            print("Ready:", name)
        # Utfør et direkte oppslag uten SQL-spørring.
        elif args.mode == "point":
            # ID og partisjonsnøkkel identifiserer dokumentet sammen.
            result = container.read_item("note-1", partition_key="containers", response_hook=report)
            # Vis dokumentets ID og tekst.
            print(result["id"], result["text"])
        # Lag en endring som change feed-triggeren kan observere.
        elif args.mode == "update":
            # Les dokumentet før oppdateringen.
            document = container.read_item("note-1", partition_key="containers")
            # Øk et synlig revisjonsnummer.
            document["revision"] = document.get("revision", 0) + 1
            # Lagre den nye versjonen og vis skrivekostnaden.
            container.upsert_item(document, response_hook=report)
            # Bruk revisjonen til å kjenne igjen endringen i triggerloggen.
            print("Updated note-1, revision", document["revision"])
        # Kjør vanlig SQL-søk eller vektorsøk.
        else:
            # Start med et metadatafilter.
            sql = "SELECT c.id, c.topic FROM c WHERE c.topic = @topic"
            # Send filterverdien som en parameter.
            parameters = [{"name": "@topic", "value": "containers"}]
            # Bytt til rangering etter vektoravstand.
            if args.mode == "vector":
                # Returner de to nærmeste treffene og deres avstand.
                sql = ("SELECT TOP 2 c.id, VectorDistance(c.embedding, @vector) AS distance "
                       # Behold emnefilteret og sorter de gjenværende dokumentene etter avstand.
                       "FROM c WHERE c.topic = @topic ORDER BY VectorDistance(c.embedding, @vector)")
                # Bruk samme tredimensjonale spørsmålsvektor som i PostgreSQL-øvelsen.
                parameters.append({"name": "@vector", "value": [0.0, 1.0, 0.0]})
            # Kjør den parameteriserte spørringen.
            results = container.query_items(sql, parameters=parameters,
                                            # Avgrens lesingen til containers-partisjonen og vis RU.
                                            partition_key="containers", response_hook=report)
            # Iterer resultatene fra SDK-et.
            for result in results:
                # Vis hvert treff i terminalen.
                print(result)


# Kjør bare når filen startes direkte.
if __name__ == "__main__":
    # Start den valgte laboperasjonen.
    main()

import argparse
# Les endpoint, nøkkel og databasenavn fra terminalens miljø.
import os

# Bruk Cosmos SDK-et og en eksplisitt partisjonsnøkkel.
from azure.cosmos import CosmosClient, PartitionKey

# Bruk små undervisningsdokumenter med håndskrevne vektorer.
DOCUMENTS = [
    # Første dokument ligger i partisjonen containers.
    {"id": "note-1", "topic": "containers", "text": "Package an application", "embedding": [1.0, 0.0, 0.0]},
    # Andre dokument ligger i samme partisjon, med en lignende vektor.
    {"id": "note-2", "topic": "containers", "text": "Inspect container logs", "embedding": [0.9, 0.1, 0.0]},
    # Tredje dokument viser hvordan metadatafilter kan utelate et nært vektortreff.
    {"id": "note-3", "topic": "data", "text": "Retrieve stored data", "embedding": [0.0, 1.0, 0.0]},
]

def report(headers, _body):
    # Hent kostnaden i request units fra responsheaderen.
    charge = headers.get("x-ms-request-charge")
    # Skriv bare RU når headeren finnes.
    if charge:
        # Vis grunnlaget for sammenligning av oppslag.
        print("Request charge:", charge, "RU")


# Kjør operasjonen som ble valgt på kommandolinjen.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Begrens valgene til operasjonene workshopene bruker.
    parser.add_argument("mode", choices=["setup", "query", "point", "vector-setup", "vector", "update"])
    # Les valgt operasjon.
    args = parser.parse_args()
    # Åpne klienten med labkontoens endpoint og nøkkel, og lukk den etter bruk.
    with CosmosClient(os.environ["COSMOS_ENDPOINT"], credential=os.environ["COSMOS_KEY"]) as client:
        # Use a database without shared throughput so vector search is supported.
        # Velg databasen; dette oppretter ikke databasen.
        database = client.get_database_client(os.getenv("COSMOS_DATABASE", "workshop"))
        # Hold vektorpolicyen i en egen container.
        name = "vectors" if args.mode.startswith("vector") else "notes"
        # Hent en klient for den valgte containeren.
        container = database.get_container_client(name)
        # Opprett labcontainer og eksempeldata ved setup.
        if args.mode in {"setup", "vector-setup"}:
            # Beskriv den vanlige indekseringspolicyen.
            policy = {
                # Oppdater indeksene automatisk når dokumentene endres.
                "indexingMode": "consistent", "automatic": True,
                # Indekser dokumentfeltene som standard.
                "includedPaths": [{"path": "/*"}],
                # Unnta vektorarrayet fra vanlig feltindeksering.
                "excludedPaths": [{"path": "/embedding/*"}],
            }
            # Legg bare til vektorinnstillinger for vektorcontaineren.
            extra = {}
            # Konfigurer en egen indeks for VectorDistance-søk.
            if args.mode == "vector-setup":
                # Bruk en flat vektorindeks på embedding-feltet.
                policy["vectorIndexes"] = [{"path": "/embedding", "type": "flat"}]
                # Beskriv hvilke vektorer embedding-feltet inneholder.
                extra["vector_embedding_policy"] = {"vectorEmbeddings": [{
                    # Lagre vektorene i embedding og behandle verdiene som float32.
                    "path": "/embedding", "dataType": "float32",
                    # Krev tre dimensjoner og sammenlign med cosinusavstand.
                    "dimensions": 3, "distanceFunction": "cosine",
                }]}
            # Opprett containeren hvis den mangler; eksisterende policy erstattes ikke her.
            container = database.create_container_if_not_exists(
                # Bruk topic som partisjonsnøkkel.
                id=name, partition_key=PartitionKey(path="/topic"),
                # Bruk policyen over og 400 RU/s dedikert throughput.
                indexing_policy=policy, offer_throughput=400, **extra,
            )
            # Skriv inn undervisningsdokumentene.
            for document in DOCUMENTS:
                # Opprett eller erstatt dokumentet, og vis operasjonens RU.
                container.upsert_item(document, response_hook=report)
            # Vis hvilken container som er klargjort.
            print("Ready:", name)
        # Utfør et direkte oppslag uten SQL-spørring.
        elif args.mode == "point":
            # ID og partisjonsnøkkel identifiserer dokumentet sammen.
            result = container.read_item("note-1", partition_key="containers", response_hook=report)
            # Vis dokumentets ID og tekst.
            print(result["id"], result["text"])
        # Lag en endring som change feed-triggeren kan observere.
        elif args.mode == "update":
            # Les dokumentet før oppdateringen.
            document = container.read_item("note-1", partition_key="containers")
            # Øk et synlig revisjonsnummer.
            document["revision"] = document.get("revision", 0) + 1
            # Lagre den nye versjonen og vis skrivekostnaden.
            container.upsert_item(document, response_hook=report)
            # Bruk revisjonen til å kjenne igjen endringen i triggerloggen.
            print("Updated note-1, revision", document["revision"])
        # Kjør vanlig SQL-søk eller vektorsøk.
        else:
            # Start med et metadatafilter.
            sql = "SELECT c.id, c.topic FROM c WHERE c.topic = @topic"
            # Send filterverdien som en parameter.
            parameters = [{"name": "@topic", "value": "containers"}]
            # Bytt til rangering etter vektoravstand.
            if args.mode == "vector":
                # Returner de to nærmeste treffene og deres avstand.
                sql = ("SELECT TOP 2 c.id, VectorDistance(c.embedding, @vector) AS distance "
                       # Behold emnefilteret og sorter de gjenværende dokumentene etter avstand.
                       "FROM c WHERE c.topic = @topic ORDER BY VectorDistance(c.embedding, @vector)")
                # Bruk samme tredimensjonale spørsmålsvektor som i PostgreSQL-øvelsen.
                parameters.append({"name": "@vector", "value": [0.0, 1.0, 0.0]})
            # Kjør den parameteriserte spørringen.
            results = container.query_items(sql, parameters=parameters,
                                            # Avgrens lesingen til containers-partisjonen og vis RU.
                                            partition_key="containers", response_hook=report)
            # Iterer resultatene fra SDK-et.
            for result in results:
                # Vis hvert treff i terminalen.
                print(result)


# Kjør bare når filen startes direkte.
if __name__ == "__main__":
    # Start den valgte laboperasjonen.
    main()

"""The Cosmos DB trigger uses the change feed processor and checkpoints in leases."""
# Send meldinger til Functions-hostens logg.
import logging
# Hent FunctionApp og dokumenttypene for Python-bindingene.
import azure.functions as func

# Opprett appen som Functions-hosten oppdager.
app = func.FunctionApp()


# Send endrede dokumenter til documents-argumentet.
# Databasenavnet hentes fra appinnstillingen COSMOS_DATABASE.
@app.cosmos_db_trigger(arg_name="documents", database_name="%COSMOS_DATABASE%",
                      # Overvåk notes med tilkoblingen fra CosmosConnection-innstillingen.
                      container_name="notes", connection="CosmosConnection",
                      # Lagre fremdrift i leases; opprett lease-containeren hvis den mangler.
                      lease_container_name="leases", create_lease_container_if_not_exists=True)
# En triggerkjøring kan inneholde en samling endrede dokumenter.
def changed(documents: func.DocumentList):
    # Behandle hvert dokument i leveransen.
    for document in documents:
        # Logg ID og revisjon slik at du kan sammenligne med oppdateringsskriptet.
        logging.info("changed id=%s revision=%s", document["id"], document.get("revision", 0))

"""Managed Redis, Enterprise clustering policy, TLS and a lab access key."""
# Velg cache-, indeks-, søke- eller oppryddingsoperasjon.
import argparse
# Les Redis-endpoint og labnøkkel fra terminalens miljø.
import os
# Vent på en kort TTL for å demonstrere utløp.
import time
# Konverter spørsmåls- og dokumentvektorer til float32-byteverdier.
from array import array

# Bruk Redis-klienten til cache- og søkekommandoer.
import redis
# Beskriv metadata, tekst og vektorer i søkeindeksen.
from redis.commands.search.field import TagField, TextField, VectorField
# Avgrens hvilke Redis-hasher indeksen leser.
from redis.commands.search.index_definition import IndexDefinition, IndexType
# Bygg et søk med filter, KNN og resultatvalg.
from redis.commands.search.query import Query
# Gjenkjenn feilsvaret når indeksen allerede finnes.
from redis.exceptions import ResponseError


# Kjør én Redis-operasjon.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Begrens operasjonene til labens fire valg.
    parser.add_argument("mode", choices=["cache", "index", "search", "cleanup"])
    # Les valgt operasjon.
    args = parser.parse_args()
    # Bruk endpoint og port for labinstansen.
    with redis.Redis(host=os.environ["REDIS_HOST"], port=int(os.getenv("REDIS_PORT", "10000")),
                     # Bruk TLS, miljøbasert labnøkkel og en avgrenset nettverkstimeout.
                     password=os.environ["REDIS_KEY"], ssl=True, socket_timeout=10) as client:
        # Demonstrer hit, invalidering og TTL-utløp.
        if args.mode == "cache":
            # Bruk en avgrenset nøkkel for labens cacheoppføring.
            key = "ai200:cache:containers:v1"
            # Kontroller om oppføringen finnes fra før.
            print("Initial:", client.get(key))
            # Cache resultatet med ti sekunders TTL.
            client.set(key, "note-1,note-2", ex=10)
            # Les den nye oppføringen og gjenværende levetid.
            print("Hit:", client.get(key), "TTL:", client.ttl(key))
            # Invalider oppføringen eksplisitt.
            client.delete(key)
            # Bekreft at oppslaget ikke lenger gir en cacheverdi.
            print("Invalidated:", client.get(key))
            # Legg inn en oppdatert verdi med kort TTL.
            client.set(key, "updated-note-1", ex=2)
            # Vent lenger enn levetiden.
            time.sleep(3)
            # Vis at Redis har latt oppføringen utløpe.
            print("Expired:", client.get(key), "TTL:", client.ttl(key))
        # Opprett en vektorindeks og tre dokumenter.
        elif args.mode == "index":
            # Indekser topic som filter, text som tekst og embedding med HNSW.
            fields = [TagField("topic"), TextField("text"), VectorField("embedding", "HNSW", {
                # Bruk float32, tre dimensjoner og cosinusavstand.
                "TYPE": "FLOAT32", "DIM": 3, "DISTANCE_METRIC": "COSINE",
            })]
            # Tillat at oppsettet kjøres flere ganger.
            try:
                # Opprett søkeindeksen med det valgte skjemaet.
                client.ft("ai200-index").create_index(fields, definition=IndexDefinition(
                    # Les bare Redis-hasher med labens vektorprefiks.
                    prefix=["ai200:vector:"], index_type=IndexType.HASH))
            # Undersøk hvorfor indeksopprettelsen ble avvist.
            except ResponseError as error:
                # Ignorer bare den forventede feilen om en eksisterende indeks.
                if "index already exists" not in str(error).lower():
                    # La andre feil gå videre til terminalen.
                    raise
            # Første dokument ligger langs første vektordimensjon.
            for number, topic, vector in [(1, "containers", [1, 0, 0]),
                                           # Legg til et lignende dokument og et dokument i et annet emne.
                                           (2, "containers", [0.9, 0.1, 0]), (3, "data", [0, 1, 0])]:
                # Lagre hvert dokument i en Redis-hash som indeksen overvåker.
                client.hset(f"ai200:vector:{number}", mapping={
                    # Lagre metadata og tekst sammen med vektoren.
                    "topic": topic, "text": f"Teaching document {number}",
                    # Send embedding som binære float32-verdier.
                    "embedding": array("f", vector).tobytes(),
                })
            # Minn om at søkeindeksen må ha rukket å behandle dataene.
            print("Index and three documents ready; wait for indexing before search")
        # Kombiner metadatafilter med vektorrangering.
        elif args.mode == "search":
            # Finn to nærmeste treff innen topic=containers, med avstanden kalt score.
            query = Query("(@topic:{containers})=>[KNN 2 @embedding $vec AS score]")
            # Sorter etter avstand og bruk søkedialekten som støtter denne syntaksen.
            query = query.sort_by("score").return_fields("topic", "text", "score").dialect(2)
            # Send spørsmålsvektoren i samme binære format som dokumentvektorene.
            result = client.ft("ai200-index").search(query, query_params={"vec": array("f", [0, 1, 0]).tobytes()})
            # Behandle de returnerte treffene.
            for document in result.docs:
                # Vis ID, emne og avstand for sammenligning.
                print(document.id, document.topic, document.score)
        # Rydd bare labens indeks og tilknyttede dokumenter.
        else:
            # Slett indeksen og dokumentene som hører til den.
            client.ft("ai200-index").dropindex(delete_documents=True)
            # Bekreft oppryddingen.
            print("Removed lab index and its documents")


# Start bare når filen kjøres direkte.
if __name__ == "__main__":
    # Kjør valgt Redis-operasjon.
    main()

2. Hit, miss og utløp

python labs/redis/demo.py cache

Forventet: først en miss, så en hit med positiv TTL, deretter miss etter invalidation og miss etter utløp. Skriptet sletter bare sin egen nøkkel. Kjør på nytt og sammenlign. Finn GET, SET med EX, DELETE og TTL i koden.

3. Knytt operasjonene til applikasjonen

Tegn rekkefølgen for cache-aside: GET, databaseoppslag ved miss, SET med TTL. Beskriv deretter hva en dokumentoppdatering skal invalidere. La nøkkelen inneholde søkeområde og dataversjon. Sammenlign kort TTL med eksplisitt invalidation; de løser ulike deler av problemet.
