-- Aktiver pgvector-utvidelsen i den valgte databasen.
CREATE EXTENSION IF NOT EXISTS vector;

-- Opprett labtabellen hvis den ikke finnes fra før.
CREATE TABLE IF NOT EXISTS workshop_notes (
    -- Gi hver rad en entydig identifikator.
    id integer PRIMARY KEY,
    -- Bruk topic som metadatafilter i søkene.
    topic text NOT NULL,
    -- Lagre teksten som et senere RAG-steg kan hente frem.
    content text NOT NULL,
    -- Lagre nøyaktig tre komponenter per undervisningsvektor.
    embedding vector(3) NOT NULL
);

-- Hand-written teaching vectors, not output from an embedding model.
-- Sett inn fire eksempler. Vektorene er skrevet for hånd, ikke generert av en modell.
INSERT INTO workshop_notes (id, topic, content, embedding) VALUES
    -- Første containerdokument peker langs første vektordimensjon.
    (1, 'containers', 'A container packages an application and its dependencies.', '[1,0,0]'),
    -- Andre containerdokument ligger nær det første i vektorrommet.
    (2, 'containers', 'Read container logs when an application fails to start.', '[0.9,0.1,0]'),
    -- Datadokumentet ligger nær spørsmålsvektoren, men kan filtreres bort av topic.
    (3, 'data', 'A database stores and retrieves application data.', '[0,1,0]'),
    -- Sikkerhetsdokumentet gir oss et eksempel i et annet emne.
    (4, 'security', 'Managed identity removes the need to store a credential.', '[0,0,1]')
-- Oppdater eksisterende rader når skriptet kjøres på nytt.
ON CONFLICT (id) DO UPDATE SET
    -- Bruk emnet fra raden vi forsøkte å sette inn.
    topic = EXCLUDED.topic,
    -- Oppdater teksten til verdien i eksempeldatasettet.
    content = EXCLUDED.content,
    -- Oppdater også vektoren, og avslutt SQL-setningen.
    embedding = EXCLUDED.embedding;

1. Koble til med Python

Start i prosjektroten. Hent tilkoblingsinformasjonen fra Connect i portalen. PGCONNINFO er en lokal miljøvariabel; behold sslmode=require og legg ikke passordet i filene.

# Sett PostgreSQL-tilkoblingen med database, bruker, passord og påkrevd TLS.
$env:PGCONNINFO = "host=DIN-SERVER.postgres.database.azure.com port=5432 dbname=workshop user=DIN-BRUKER password=DITT-PASSORD sslmode=require"
# Opprett labtabellen, aktiver vector og last inn de fire eksempelradene i Azure-databasen.
python labs/data/postgres_client.py setup
# Kjør den parameteriserte PostgreSQL-spørringen og vis de nærmeste dokumentene.
python labs/data/postgres_client.py query

Forventet: dokument 2 og 1. Finn de parameteriserte verdiene i postgres_client.py og forklar hvorfor SQL og data sendes separat.

postgres_client.py — vis fil og forklaring

Kildefil: labs/data/postgres_client.py

"""Run the same parameterized vector query with a direct connection and a pool."""
# Velg setup, query eller compare fra kommandolinjen.
import argparse
# Les tilkoblingen fra miljøet fremfor å lagre passord i filen.
import os
# Mål tiden brukt på direkte forbindelser og forbindelsespool.
import time
# Finn SQL-filen ved siden av dette skriptet.
from pathlib import Path

# Bruk PostgreSQL-driveren til å kjøre SQL.
import psycopg
# Registrer hvordan pgvector-verdier konverteres i forbindelsen.
from pgvector.psycopg import register_vector
# Gjenbruk et lite antall databaseforbindelser.
from psycopg_pool import ConnectionPool


# Kjør dette oppsettet på hver nye forbindelse i poolen.
def configure(connection):
    # Lær forbindelsen å håndtere databasens vector-type.
    register_vector(connection)
    # Avslutt transaksjonen fra typeoppsettet før forbindelsen brukes av poolen.
    connection.commit()


# Bruk samme spørring i begge målingene.
def query(connection):
    # Kjør parameterisert SQL og hent de to nærmeste radene.
    return connection.execute(
        # Hent dokumentets ID, metadata og tekst.
        "SELECT id, topic, content FROM workshop_notes "
        # Filtrer på emne og sorter etter cosinusavstand med pgvector.
        "WHERE topic = %s ORDER BY embedding <=> %s::vector LIMIT 2",
        # Send verdiene separat fra SQL-teksten.
        ("containers", "[0,1,0]"),
    ).fetchall()


# Samle oppsett, spørring og sammenligning i ett kjørbart eksempel.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Tillat bare de tre laboperasjonene.
    parser.add_argument("mode", choices=["setup", "query", "compare"])
    # Les operasjonen deltakeren valgte.
    args = parser.parse_args()
    # Bruk tilkoblingsstrengen satt i terminalen.
    connection_string = os.environ["PGCONNINFO"]
    # Disable prepared statements so the example also works through transaction pooling.
    # Deaktiver automatisk server-side prepare for å støtte transaction pooling.
    options = {"prepare_threshold": None}
    # Opprett skjema og eksempeldata.
    if args.mode == "setup":
        # Åpne og lukk en direkte forbindelse for denne operasjonen.
        with psycopg.connect(connection_string, **options) as connection:
            # Les setup.sql fra samme mappe og kjør innholdet på serveren.
            connection.execute(Path(__file__).with_name("setup.sql").read_text(encoding="utf-8"))
        # Bekreft at oppsettet er fullført.
        print("Schema and four example rows ready")
    # Kjør ett vanlig oppslag.
    elif args.mode == "query":
        # Åpne og lukk en direkte forbindelse for denne operasjonen.
        with psycopg.connect(connection_string, **options) as connection:
            # Behandle radene fra den parameteriserte spørringen.
            for row in query(connection):
                # Vis resultatet i terminalen.
                print(row)
    # Sammenlign tidsbruk med og uten forbindelsespool.
    else:
        # Bruk samme antall operasjoner i begge målingene.
        count = 10
        # Start tidsmålingen for direkte forbindelser.
        started = time.perf_counter()
        # Åpne en ny forbindelse for hver repetisjon.
        for _ in range(count):
            # Inkluder forbindelsesopprettelsen i målingen.
            with psycopg.connect(connection_string, **options) as connection:
                # Kjør samme arbeid som i pool-varianten.
                query(connection)
        # Vis samlet tid for nye forbindelser.
        print(f"{count} new connections: {time.perf_counter() - started:.3f}s")
        # Opprett en pool med mellom én og tre forbindelser.
        with ConnectionPool(connection_string, min_size=1, max_size=3,
                            # Bruk samme tilkoblingsvalg og registrer pgvector for hver forbindelse.
                            kwargs=options, configure=configure) as pool:
            # Vent til poolen er klar før målingen starter.
            pool.wait()
            # Start målingen etter klargjøringen av poolen.
            started = time.perf_counter()
            # Gjenta samme antall spørringer.
            for _ in range(count):
                # Lån en forbindelse og lever den tilbake etter bruk.
                with pool.connection() as connection:
                    # Kjør den samme spørringen på en gjenbrukt forbindelse.
                    query(connection)
            # Vis tiden for spørringene gjennom poolen.
            print(f"{count} pooled queries: {time.perf_counter() - started:.3f}s")


# Start bare når filen kjøres direkte.
if __name__ == "__main__":
    # Kjør den valgte laboperasjonen.
    main()

2. Mål og gjenbruk forbindelser

# Mål ti nye forbindelser og sammenlign med ti oppslag gjennom en klientpool.
python labs/data/postgres_client.py compare

Skriptet sammenligner ti nye forbindelser med ti spørringer via en liten klientpool. Gjenta og noter variasjonen. Tilkoblingslatens, nettverk og oppvarming påvirker resultatene. Undersøk min_size/max_size og hvorfor hver prosess trenger sin egen pool.

3. Undersøk Azure-flaskehalser

  1. Åpne Metrics for CPU, minne, aktive forbindelser og lagrings-I/O. Kjør compare igjen og noter hva en så liten last faktisk kan vise.
  2. Undersøk Compute + storage og PgBouncer. Velg et foreslått alternativ for mange samtidige oppslag og begrunn CPU/minne/lagring. Gjennomfør en endring bare på din labserver hvis øvelsen trenger den.
  3. På en tier med innebygd PgBouncer: aktiver den, bruk port 6432 i PGCONNINFO og kjør query/compare på nytt. Hvis tier-en ikke støtter det, bruker du klientpoolen og forklarer skillet.

Kombiner målingene med EXPLAIN-planen fra workshop 7. En liten datasamling er ikke en representativ kapasitetsmåling.
