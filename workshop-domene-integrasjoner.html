"""Local workshop API. In-memory deduplication is deliberately not production storage."""

# Serialiser svarene som JSON.
import json
# Skriv hendelser og kontrollerte feil til Functions-loggen.
import logging
# Les appinnstillingene fra miljøet.
import os
# Beskytt det delte minnesettet mot samtidige forespørsler.
import threading

# Bruk Functions-dekoratorer og HTTP-typer.
import azure.functions as func

# Krev en Function-nøkkel på vanlige HTTP-ruter i Azure.
app = func.FunctionApp(http_auth_level=func.AuthLevel.FUNCTION)
# Husk behandlede dokument-ID-er bare så lenge denne prosessen kjører.
processed = set()
# La kontroll og registrering av en ID skje samlet.
processed_lock = threading.Lock()


# Ta imot dokumenter på /api/documents.
@app.route(route="documents", methods=["POST"])
# Valider dokumentet og demonstrer håndtering av samme ID flere ganger.
def documents(req: func.HttpRequest) -> func.HttpResponse:
    # Forsøk å lese forespørselsinnholdet som JSON.
    try:
        # Konverter JSON-body til Python-data.
        payload = req.get_json()
    # Håndter innhold som ikke kan tolkes som JSON.
    except ValueError:
        # Svar med klientfeil for ugyldig JSON.
        return func.HttpResponse("Body must be JSON", status_code=400)
    # Krev et JSON-objekt, ikke for eksempel en liste.
    if not isinstance(payload, dict):
        # Forklar hvorfor datatypen ble avvist.
        return func.HttpResponse("Body must be a JSON object", status_code=400)
    # Hent dokument-ID-en som brukes til duplikatkontroll.
    document_id = payload.get("id")
    # Avvis manglende, tomme og ikke-tekstlige ID-er.
    if not isinstance(document_id, str) or not document_id.strip():
        # Send en tydelig valideringsfeil til klienten.
        return func.HttpResponse("id must be a non-empty string", status_code=400)
    # Unngå at to samtidige kall registrerer samme ID som ny.
    with processed_lock:
        # Kontroller om denne prosessen allerede har sett ID-en.
        duplicate = document_id in processed
        # Registrer ID-en før en annen tråd får tilgang.
        processed.add(document_id)
    # Gjør duplikathåndteringen synlig i loggen.
    logging.info("document_received duplicate=%s", duplicate)
    # Returner et JSON-svar til klienten.
    return func.HttpResponse(
        # Vis om dette kallet ble behandlet som nytt.
        json.dumps({"id": document_id, "processed": not duplicate}),
        # Angi riktig innholdstype for JSON-responsen.
        mimetype="application/json",
    )


# La Key Vault-øvelsen kontrollere om hemmeligheten ble levert til appen.
@app.route(route="configuration", methods=["GET"])
# Rapporter bare status for konfigurasjonen.
def configuration(req: func.HttpRequest) -> func.HttpResponse:
    # Les hemmeligheten som appinnstilling.
    secret = os.getenv("DEMO_SECRET", "")
    # Skill en faktisk verdi fra en Key Vault-referanse som ikke er løst.
    resolved = bool(secret) and not secret.startswith("@Microsoft.KeyVault(")
    # Returner et JSON-svar til klienten.
    return func.HttpResponse(
        # Send bare en boolsk status, aldri hemmelighetens verdi.
        json.dumps({"secret_configured": resolved}),
        # Angi riktig innholdstype for JSON-responsen.
        mimetype="application/json",
    )


# Gi labens Event Grid-webhook en HTTP-rute uten Function-nøkkel.
@app.route(route="event-hook", methods=["POST"], auth_level=func.AuthLevel.ANONYMOUS)
# Håndter valideringshendelser og vanlige Event Grid-hendelser.
def event_hook(req: func.HttpRequest) -> func.HttpResponse:
    # Les hendelseslisten i Event Grid Schema-format.
    events = req.get_json()
    # Krev en liste slik dette hendelsesskjemaet bruker.
    if not isinstance(events, list):
        # Avvis et annet meldingsformat.
        return func.HttpResponse("Expected Event Grid Schema array", status_code=400)
    # Gå gjennom hendelsene i leveransen.
    for event in events:
        # Gjenkjenn håndtrykket som validerer webhook-abonnementet.
        if event.get("eventType") == "Microsoft.EventGrid.SubscriptionValidationEvent":
            # Send valideringskoden tilbake i validationResponse.
            return func.HttpResponse(json.dumps({"validationResponse": event["data"]["validationCode"]}),
                                     # Svar med JSON slik valideringen forventer.
                                     mimetype="application/json")
    # Bruk en appinnstilling til å slå på en kontrollert leveringsfeil.
    if os.getenv("FAIL_EVENTS", "false").lower() == "true":
        # Gjør den fremprovoserte feilen synlig i loggen.
        logging.warning("Controlled Event Grid delivery failure")
        # Svar med 503 så Event Grid kan forsøke levering igjen.
        return func.HttpResponse("Retry this delivery", status_code=503)
    # Gå gjennom hendelsene i leveransen.
    for event in events:
        # Vis hvilke hendelser som kom gjennom abonnementets filter.
        logging.info("event id=%s subject=%s", event.get("id"), event.get("subject"))
    # Bekreft vellykket mottak med HTTP 200.
    return func.HttpResponse("Accepted")

message_bus.py — vis fil og forklaring

Arbeidskopi: artifacts/labs/integrations/message_bus.py

"""Service Bus data-plane operations using the signed-in user's identity."""
# Velg kø-, topic-, subscription- eller DLQ-operasjon.
import argparse
# Les Service Bus-navnet fra miljøet.
import os

# Bruk Entra-identitet, for eksempel den innloggede Azure CLI-brukeren.
from azure.identity import DefaultAzureCredential
# Hent klienten, meldingstypen og valget for dead-letter-køen.
from azure.servicebus import ServiceBusClient, ServiceBusMessage, ServiceBusSubQueue


# Kjør én operasjon mot labnamespace-et.
def main():
    # Opprett kommandolinjeparseren.
    parser = argparse.ArgumentParser()
    # Begrens operasjonene til dem som brukes i øvelsene.
    parser.add_argument("mode", choices=["send", "receive", "dlq", "topic", "subscription"])
    # Velg mottakende subscription; indexer er standard.
    parser.add_argument("--subscription", default="indexer")
    # Les valgene fra terminalen.
    args = parser.parse_args()
    # Opprett identitet og klient, og lukk dem etter operasjonen.
    with DefaultAzureCredential() as credential, ServiceBusClient(
            # Bruk namespace-adressen uten å lagre en tilkoblingsnøkkel i filen.
            fully_qualified_namespace=os.environ["SERVICEBUS_NAMESPACE"], credential=credential) as client:
        # Disse operasjonene sender én ny melding.
        if args.mode in {"send", "topic"}:
            # Send til topic-et når topic-operasjonen er valgt.
            sender = (client.get_topic_sender("document-events") if args.mode == "topic"
                      # Ellers sendes meldingen til documents-køen.
                      else client.get_queue_sender("documents"))
            # Åpne senderen for denne sendingen.
            with sender:
                # Bruk en liten JSON-body som nyttelast.
                sender.send_messages(ServiceBusMessage('{"id":"doc-001"}',
                                                       # Legg til subject og en egenskap som subscription-filtre kan bruke.
                                                       subject="document", application_properties={"kind": "note"}))
            # Bekreft at SDK-sendingen er fullført.
            print("Sent")
        # De øvrige operasjonene mottar meldinger.
        else:
            # En topic-mottaker leser fra en bestemt subscription.
            if args.mode == "subscription":
                # Velg indexer eller audit på document-events-topic-et.
                receiver = client.get_subscription_receiver("document-events", args.subscription)
            # Køøvelsen leser enten vanlig kø eller dens DLQ.
            else:
                # Opprett mottakeren for documents-køen.
                receiver = client.get_queue_receiver("documents", sub_queue=(
                    # Velg dead-letter-underkøen bare i dlq-modus.
                    ServiceBusSubQueue.DEAD_LETTER if args.mode == "dlq" else None))
            # Åpne mottakeren og lukk den etter bruk.
            with receiver:
                # Hent høyst ti meldinger og begrens ventingen til fem sekunder.
                messages = receiver.receive_messages(max_message_count=10, max_wait_time=5)
                # Vis hvor mange meldinger som ble hentet.
                print("Received:", len(messages))
                # Behandle hver mottatt melding.
                for message in messages:
                    # Vis body og antall leveringsforsøk.
                    print(str(message), "Delivery:", message.delivery_count)
                    # Fremprovoser DLQ-flyten i den vanlige køøvelsen.
                    if args.mode == "receive":
                        # Flytt meldingen til DLQ med en årsak du kan inspisere.
                        receiver.dead_letter_message(message, reason="Workshop", error_description="Controlled demo")
                    # Fullfør meldinger i DLQ- og subscription-øvelsene.
                    else:
                        # Bekreft behandling og fjern meldingen fra denne mottakerkøen.
                        receiver.complete_message(message)


# Start bare når filen kjøres direkte.
if __name__ == "__main__":
    # Kjør valgt Service Bus-operasjon.
    main()

"""HTTP -> Service Bus output binding -> Service Bus trigger."""
# Skriv behandlingsinformasjon til Functions-loggen.
import logging
# Bruk Functions-dekoratorer, HTTP-typer og bindingstyper.
import azure.functions as func

# Krev en Function-nøkkel på HTTP-rutene når appen kjører i Azure.
app = func.FunctionApp(http_auth_level=func.AuthLevel.FUNCTION)


# Ta imot HTTP POST på /api/enqueue.
@app.route(route="enqueue", methods=["POST"])
# Koble message-outputen til documents-køen med den navngitte appinnstillingen.
@app.service_bus_queue_output(arg_name="message", queue_name="documents", connection="ServiceBusConnection")
# Motta HTTP-kallet og et output-objekt som Functions-hosten håndterer.
def enqueue(req: func.HttpRequest, message: func.Out[str]) -> func.HttpResponse:
    # Legg HTTP-body i output-bindingen; hosten står for sendingen.
    message.set(req.get_body().decode("utf-8"))
    # Svar at forespørselen er akseptert.
    return func.HttpResponse("Queued", status_code=202)


# Start denne funksjonen når en melding kan behandles fra documents-køen.
@app.service_bus_queue_trigger(arg_name="message", queue_name="documents", connection="ServiceBusConnection")
# Motta meldingen og metadataene fra triggeren.
def consume(message: func.ServiceBusMessage):
    # Les meldingsteksten som UTF-8.
    body = message.get_body().decode("utf-8")
    # Bruk en bestemt tekst til å fremprovosere behandlingsfeil.
    if "fail" in body.lower():
        # La feilen gå til hosten så ny levering og DLQ kan observeres.
        raise ValueError("Controlled failure for delivery-count and DLQ exercise")
    # Vis meldingens ID og antall leveringsforsøk ved vellykket behandling.
    logging.info("Consumed id=%s delivery=%s", message.message_id, message.delivery_count)

"""Local workshop API. In-memory deduplication is deliberately not production storage."""

# Serialiser svarene som JSON.
import json
# Skriv hendelser og kontrollerte feil til Functions-loggen.
import logging
# Les appinnstillingene fra miljøet.
import os
# Beskytt det delte minnesettet mot samtidige forespørsler.
import threading

# Bruk Functions-dekoratorer og HTTP-typer.
import azure.functions as func

# Krev en Function-nøkkel på vanlige HTTP-ruter i Azure.
app = func.FunctionApp(http_auth_level=func.AuthLevel.FUNCTION)
# Husk behandlede dokument-ID-er bare så lenge denne prosessen kjører.
processed = set()
# La kontroll og registrering av en ID skje samlet.
processed_lock = threading.Lock()


# Ta imot dokumenter på /api/documents.
@app.route(route="documents", methods=["POST"])
# Valider dokumentet og demonstrer håndtering av samme ID flere ganger.
def documents(req: func.HttpRequest) -> func.HttpResponse:
    # Forsøk å lese forespørselsinnholdet som JSON.
    try:
        # Konverter JSON-body til Python-data.
        payload = req.get_json()
    # Håndter innhold som ikke kan tolkes som JSON.
    except ValueError:
        # Svar med klientfeil for ugyldig JSON.
        return func.HttpResponse("Body must be JSON", status_code=400)
    # Krev et JSON-objekt, ikke for eksempel en liste.
    if not isinstance(payload, dict):
        # Forklar hvorfor datatypen ble avvist.
        return func.HttpResponse("Body must be a JSON object", status_code=400)
    # Hent dokument-ID-en som brukes til duplikatkontroll.
    document_id = payload.get("id")
    # Avvis manglende, tomme og ikke-tekstlige ID-er.
    if not isinstance(document_id, str) or not document_id.strip():
        # Send en tydelig valideringsfeil til klienten.
        return func.HttpResponse("id must be a non-empty string", status_code=400)
    # Unngå at to samtidige kall registrerer samme ID som ny.
    with processed_lock:
        # Kontroller om denne prosessen allerede har sett ID-en.
        duplicate = document_id in processed
        # Registrer ID-en før en annen tråd får tilgang.
        processed.add(document_id)
    # Gjør duplikathåndteringen synlig i loggen.
    logging.info("document_received duplicate=%s", duplicate)
    # Returner et JSON-svar til klienten.
    return func.HttpResponse(
        # Vis om dette kallet ble behandlet som nytt.
        json.dumps({"id": document_id, "processed": not duplicate}),
        # Angi riktig innholdstype for JSON-responsen.
        mimetype="application/json",
    )


# La Key Vault-øvelsen kontrollere om hemmeligheten ble levert til appen.
@app.route(route="configuration", methods=["GET"])
# Rapporter bare status for konfigurasjonen.
def configuration(req: func.HttpRequest) -> func.HttpResponse:
    # Les hemmeligheten som appinnstilling.
    secret = os.getenv("DEMO_SECRET", "")
    # Skill en faktisk verdi fra en Key Vault-referanse som ikke er løst.
    resolved = bool(secret) and not secret.startswith("@Microsoft.KeyVault(")
    # Returner et JSON-svar til klienten.
    return func.HttpResponse(
        # Send bare en boolsk status, aldri hemmelighetens verdi.
        json.dumps({"secret_configured": resolved}),
        # Angi riktig innholdstype for JSON-responsen.
        mimetype="application/json",
    )


# Gi labens Event Grid-webhook en HTTP-rute uten Function-nøkkel.
@app.route(route="event-hook", methods=["POST"], auth_level=func.AuthLevel.ANONYMOUS)
# Håndter valideringshendelser og vanlige Event Grid-hendelser.
def event_hook(req: func.HttpRequest) -> func.HttpResponse:
    # Les hendelseslisten i Event Grid Schema-format.
    events = req.get_json()
    # Krev en liste slik dette hendelsesskjemaet bruker.
    if not isinstance(events, list):
        # Avvis et annet meldingsformat.
        return func.HttpResponse("Expected Event Grid Schema array", status_code=400)
    # Gå gjennom hendelsene i leveransen.
    for event in events:
        # Gjenkjenn håndtrykket som validerer webhook-abonnementet.
        if event.get("eventType") == "Microsoft.EventGrid.SubscriptionValidationEvent":
            # Send valideringskoden tilbake i validationResponse.
            return func.HttpResponse(json.dumps({"validationResponse": event["data"]["validationCode"]}),
                                     # Svar med JSON slik valideringen forventer.
                                     mimetype="application/json")
    # Bruk en appinnstilling til å slå på en kontrollert leveringsfeil.
    if os.getenv("FAIL_EVENTS", "false").lower() == "true":
        # Gjør den fremprovoserte feilen synlig i loggen.
        logging.warning("Controlled Event Grid delivery failure")
        # Svar med 503 så Event Grid kan forsøke levering igjen.
        return func.HttpResponse("Retry this delivery", status_code=503)
    # Gå gjennom hendelsene i leveransen.
    for event in events:
        # Vis hvilke hendelser som kom gjennom abonnementets filter.
        logging.info("event id=%s subject=%s", event.get("id"), event.get("subject"))
    # Bekreft vellykket mottak med HTTP 200.
    return func.HttpResponse("Accepted")

"""Publish custom Event Grid events to a dedicated workshop topic."""
# Les topic-endpoint og labnøkkel fra miljøet.
import os
# Pakk inn Event Grid-nøkkelen i SDK-ets credential-type.
from azure.core.credentials import AzureKeyCredential
# Bruk Event Grid Schema-hendelser og publisher-klienten.
from azure.eventgrid import EventGridEvent, EventGridPublisherClient


# Publiser to hendelser for å teste abonnementets filter.
def main():
    # Koble publisher-klienten til ditt eget labtopic.
    client = EventGridPublisherClient(os.environ["EVENTGRID_ENDPOINT"],
                                     # Les tilgangen fra terminalen; ingen nøkkel lagres i kodefilen.
                                     AzureKeyCredential(os.environ["EVENTGRID_KEY"]))
    # Lukk nettverksklienten etter sendingen.
    with client:
        # Første hendelse har subject som matcher documents/ og .txt-filteret.
        client.send([EventGridEvent(subject="documents/note.txt", event_type="AI200.DocumentCreated",
                                    # Versjoner nyttelastskjemaet og send en liten dokument-ID.
                                    data_version="1.0", data={"id": "doc-001"}),
                     # Andre hendelse har samme type, men en annen filendelse.
                     EventGridEvent(subject="documents/note.csv", event_type="AI200.DocumentCreated",
                                    # Send begge i samme operasjon; abonnementet velger hvilke som leveres.
                                    data_version="1.0", data={"id": "doc-002"})])
    # Sammenlign bekreftelsen med hendelsene i webhook-loggen.
    print("Published two events; the .txt subject filter should deliver only the first")


# Start bare når filen kjøres direkte.
if __name__ == "__main__":
    # Send labhendelsene.
    main()
